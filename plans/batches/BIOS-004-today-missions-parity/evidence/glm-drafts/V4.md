import { useEffect } from 'react';
import type { ReactElement } from 'react';
import { StyleSheet, View } from 'react-native';
import Animated, {
  Easing,
  useAnimatedStyle,
  useSharedValue,
  withRepeat,
  withTiming,
} from 'react-native-reanimated';
import { LinearGradient } from 'expo-linear-gradient';

import { useReducedMotion } from '@/components/balencia/auth-inputs';
import { BalenciaColors } from '@/constants/theme';
import { MotionBreathe } from '@/constants/motion';

export interface CIAPresenceOrbProps {
  state?: 'idle' | 'listening' | 'thinking';
  size?: number;
  testID?: string;
}

export function CIAPresenceOrb({
  state = 'idle',
  size = 72,
  testID,
}: CIAPresenceOrbProps): ReactElement {
  const reduceMotion = useReducedMotion();

  const breatheScale = useSharedValue<number>(1);
  const breatheOpacity = useSharedValue<number>(MotionBreathe.opacityMin);
  const ringScale = useSharedValue<number>(1);
  const ringOpacity = useSharedValue<number>(0);
  const shimmerRotate = useSharedValue<number>(0);

  useEffect(() => {
    if (reduceMotion) {
      breatheScale.value = 1;
      breatheOpacity.value = MotionBreathe.opacityMax;
      ringScale.value = 1;
      ringOpacity.value = 0;
      shimmerRotate.value = 0;
      return;
    }

    if (state === 'idle') {
      breatheScale.value = withRepeat(
        withTiming(1.04, {
          duration: MotionBreathe.durationMs,
          easing: Easing.inOut(Easing.ease),
        }),
        -1,
        true,
      );
      breatheOpacity.value = withRepeat(
        withTiming(MotionBreathe.opacityMax, {
          duration: MotionBreathe.durationMs,
          easing: Easing.inOut(Easing.ease),
        }),
        -1,
        true,
      );
    } else if (state === 'listening') {
      breatheScale.value = 1;
      breatheOpacity.value = 1;

      ringScale.value = withRepeat(
        withTiming(1.4, { duration: 1400, easing: Easing.out(Easing.ease) }),
        -1,
        false,
      );
      ringOpacity.value = withRepeat(
        withTiming(0, { duration: 1400, easing: Easing.out(Easing.ease) }),
        -1,
        false,
      );
    } else if (state === 'thinking') {
      breatheScale.value = 1;
      breatheOpacity.value = 1;

      shimmerRotate.value = withRepeat(
        withTiming(360, { duration: 2400, easing: Easing.linear }),
        -1,
        false,
      );
    }
  }, [reduceMotion, state, breatheScale, breatheOpacity, ringScale, ringOpacity, shimmerRotate]);

  const haloAnimatedStyle = useAnimatedStyle(() => {
    return {
      opacity: breatheOpacity.value,
      transform: [{ scale: breatheScale.value }],
    };
  });

  const ringAnimatedStyle = useAnimatedStyle(() => {
    return {
      opacity: ringOpacity.value,
      transform: [{ scale: ringScale.value }],
    };
  });

  const shimmerAnimatedStyle = useAnimatedStyle(() => {
    return {
      transform: [{ rotate: `${shimmerRotate.value}deg` }],
    };
  });

  const coreSize = size * 0.52;
  const gradientColors = [
    BalenciaColors.purpleText,
    BalenciaColors.purple,
  ] as const;

  return (
    <View
      accessibilityRole="image"
      accessibilityLabel={`CIA presence, ${state}`}
      testID={testID}
      style={[styles.root, { width: size, height: size }]}
    >
      <View
        style={[
          styles.outerWash,
          {
            width: size,
            height: size,
            borderRadius: size / 2,
            backgroundColor: BalenciaColors.purpleSoft,
          },
        ]}
      />

      <View
        style={[
          styles.midHaloBase,
          {
            width: size * 0.8,
            height: size * 0.8,
            borderRadius: (size * 0.8) / 2,
          },
        ]}
      >
        <View style={[StyleSheet.absoluteFill, styles.haloBaseColor]} />
        <Animated.View style={[StyleSheet.absoluteFill, styles.haloOverlay, haloAnimatedStyle]} />
      </View>

      <Animated.View
        style={[
          styles.ringBase,
          {
            width: size * 0.68,
            height: size * 0.68,
            borderRadius: (size * 0.68) / 2,
          },
          ringAnimatedStyle,
        ]}
      />

      <View
        style={[
          styles.coreContainer,
          {
            width: coreSize,
            height: coreSize,
            borderRadius: coreSize / 2,
          },
        ]}
      >
        <LinearGradient
          colors={gradientColors}
          start={{ x: 0.3, y: 0.3 }}
          end={{ x: 1, y: 1 }}
          style={styles.gradient}
        />

        <Animated.View
          style={[
            styles.shimmerOverlay,
            {
              width: coreSize,
              height: coreSize,
              borderRadius: coreSize / 2,
            },
            shimmerAnimatedStyle,
          ]}
        >
          <LinearGradient
            colors={['rgba(255, 255, 255, 0.5)', 'transparent', 'transparent']}
            start={{ x: 0, y: 0 }}
            end={{ x: 1, y: 0 }}
            style={styles.gradient}
          />
        </Animated.View>

        <View
          style={[
            styles.specular,
            {
              width: coreSize * 0.25,
              height: coreSize * 0.25,
              borderRadius: (coreSize * 0.25) / 2,
            },
          ]}
        />
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  root: {
    alignItems: 'center',
    justifyContent: 'center',
  },
  outerWash: {
    position: 'absolute',
  },
  midHaloBase: {
    position: 'absolute',
    alignItems: 'center',
    justifyContent: 'center',
  },
  haloBaseColor: {
    backgroundColor: BalenciaColors.purpleSoft,
  },
  haloOverlay: {
    backgroundColor: BalenciaColors.purple,
  },
  ringBase: {
    position: 'absolute',
    borderWidth: 2,
    borderColor: BalenciaColors.purple,
  },
  coreContainer: {
    alignItems: 'center',
    justifyContent: 'center',
    shadowColor: BalenciaColors.purple,
    shadowOpacity: 0.36,
    shadowRadius: 10,
    shadowOffset: { width: 0, height: 0 },
    overflow: 'hidden',
  },
  gradient: {
    width: '100%',
    height: '100%',
  },
  shimmerOverlay: {
    position: 'absolute',
    overflow: 'hidden',
  },
  specular: {
    position: 'absolute',
    top: '20%',
    left: '20%',
    backgroundColor: BalenciaColors.paper100,
    opacity: 0.8,
  },
});
