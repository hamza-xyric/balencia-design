const CELEBRATION_BREATHE = { durationMs: 3000, opacityMin: 0.6, opacityMax: 1 } as const;

function CelebrationBadge({ reduceMotion }: { reduceMotion: boolean }): JSX.Element {
  const scale = useSharedValue(reduceMotion ? 1 : 0);

  useEffect(() => {
    if (reduceMotion) {
      scale.value = 1;
      return;
    }
    scale.value = withSpring(1, MotionSpring.badge);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [reduceMotion]);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  return (
    <Animated.View
      style={[styles.badgeWrapper, animatedStyle]}
      accessible
      accessibilityLabel="Mission complete"
    >
      <ProgressRing
        progress={100}
        glow="done"
        glyph={<Text style={styles.badgeGlyph}>{'✓'}</Text>}
        testID="celebration-badge-ring"
      />
    </Animated.View>
  );
}

export function CelebrationScreen(): JSX.Element {
  const router = useRouter();
  const params = useLocalSearchParams<{ xpDelta?: string | string[]; title?: string | string[] }>();
  const reduceMotion = useReducedMotion();

  const xpDelta = parseXpDelta(params.xpDelta);
  const missionTitle = titleForDisplay(params.title);

  const translateY = useSharedValue(reduceMotion ? 0 : 24);
  const opacity = useSharedValue(reduceMotion ? 1 : 0);

  useEffect(() => {
    // Badge-snap haptic (scope-hifi S42 Motion: "medium-weight impact synced
    // to badge scale-in snap"). Canon: "text updates and haptics still
    // occur" under reduced motion — fires unconditionally.
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium).catch(() => undefined);

    if (reduceMotion) {
      translateY.value = 0;
      opacity.value = 1;
      return;
    }
    const easing = Easing.bezier(...MotionEasing.physical);
    translateY.value = withTiming(0, { duration: MotionDuration.toastIn, easing });
    opacity.value = withTiming(1, { duration: MotionDuration.toastIn, easing });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [reduceMotion]);

  const goBack = () => {
    router.back();
  };

  const handleContinue = () => {
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => undefined);

    if (reduceMotion) {
      goBack();
      return;
    }
    const easing = Easing.bezier(...MotionEasing.physical);
    translateY.value = withTiming(24, { duration: MotionDuration.toastOut, easing });
    opacity.value = withTiming(0, { duration: MotionDuration.toastOut, easing }, (finished) => {
      if (finished) {
        runOnJS(goBack)();
      }
    });
  };

  const containerAnimatedStyle = useAnimatedStyle(() => ({
    opacity: opacity.value,
    transform: [{ translateY: translateY.value }],
  }));

  return (
    <View style={styles.scrim} testID="celebration-screen">
      <SafeAreaView edges={['top', 'bottom']} style={styles.safeArea}>
        <Animated.View style={[styles.cardWrapper, containerAnimatedStyle]}>
          <FrostCard glow="done" breathe breatheConfig={CELEBRATION_BREATHE} testID="celebration-card">

              </View>

// Pure, framework-free parsing for the S42 celebration route's search params
// (BIOS-004 ADR-4 XP honesty + this packet's DEFENSIVE HONESTY requirement).
//
// The S3 mission-completion mutation (`use-mission-mutations.ts`) pushes this
// route ONLY after a REFETCHED gamification-stats snapshot confirms a
// positive XP delta (`CompleteMissionResult.xpDelta: number | null`,
// stringified for the router.push params). This module still treats both
// params as untrusted at the boundary — expo-router's `useLocalSearchParams`
// always returns `string | string[] | undefined` at runtime regardless of
// the TS generic passed to it (a deep link with a repeated query key, e.g.
// `balencia://celebration?xpDelta=1&xpDelta=2`, produces an array even
// though the app's own navigation never does) — so a garbled or absent
// param must resolve to the honest-null branch, never a fabricated number.

export function parseXpDelta(param: string | string[] | undefined): number | null {
  const raw = Array.isArray(param) ? param[0] : param;
  if (raw === undefined) {
    return null;
  }
  const trimmed = raw.trim();
  // Server-confirmed XP deltas are always positive integers (ADR-4) — accept
  // ONLY a plain digit string. No leading '+', no decimals, no scientific
  // notation, no embedded whitespace: any of those are garbled input, not
  // coerced.
  if (!/^\d+$/.test(trimmed)) {
    return null;
  }
  const value = Number(trimmed);
  if (!Number.isFinite(value) || value <= 0) {
    return null;
  }
  return value;
}

const TITLE_MAX_LENGTH = 80;

export function titleForDisplay(param: string | string[] | undefined): string | null {
  const raw = Array.isArray(param) ? param[0] : param;
  if (raw === undefined) {
    return null;
  }
  const trimmed = raw.trim();
  if (trimmed.length === 0) {
    return null;
  }
  if (trimmed.length <= TITLE_MAX_LENGTH) {
    return trimmed;
  }
  return `${trimmed.slice(0, TITLE_MAX_LENGTH - 3)}...`;
}

import { describe, expect, it } from 'vitest';

import { parseXpDelta, titleForDisplay } from '@/features/missions/celebration-logic';

describe('parseXpDelta (S42 entry-param defensive honesty, ADR-4)', () => {
  it('parses a valid positive-integer string', () => {
    expect(parseXpDelta('120')).toBe(120);
  });

  it('rejects zero — never a "you earned 0 XP" fabrication', () => {
    expect(parseXpDelta('0')).toBeNull();
  });

  it('rejects a negative value', () => {
    expect(parseXpDelta('-5')).toBeNull();
  });

  it('rejects a non-numeric string', () => {
    expect(parseXpDelta('abc')).toBeNull();
  });

  it('rejects a decimal value (server deltas are always integers)', () => {
    expect(parseXpDelta('120.5')).toBeNull();
  });

  it('returns null for an absent param', () => {
    expect(parseXpDelta(undefined)).toBeNull();
  });

  it('uses the FIRST element when expo-router hands back an array (repeated query key)', () => {
    expect(parseXpDelta(['12', '3'])).toBe(12);
  });

  it('rejects an array whose first element is garbled, even if a later element is valid', () => {
    expect(parseXpDelta(['abc', '12'])).toBeNull();
  });
});

describe('titleForDisplay (S3 push-origin sanitize, defensive against deep-link tampering)', () => {
  it('trims surrounding whitespace', () => {
    expect(titleForDisplay('  Morning Sunlight  ')).toBe('Morning Sunlight');
  });

  it('returns null for an empty string', () => {
    expect(titleForDisplay('')).toBeNull();
  });

  it('returns null for a whitespace-only string', () => {
    expect(titleForDisplay('   ')).toBeNull();
  });

  it('returns null for an absent param', () => {
    expect(titleForDisplay(undefined)).toBeNull();
  });

  it('passes through a string at or under 80 chars unchanged', () => {
    const exact80 = 'x'.repeat(80);
    expect(titleForDisplay(exact80)).toBe(exact80);
  });

  it('caps a string over 80 chars to 80 total characters with an ellipsis', () => {
    const long = 'x'.repeat(120);
    const result = titleForDisplay(long);
    expect(result).toHaveLength(80);
    expect(result?.endsWith('...')).toBe(true);
  });

  it('uses the FIRST element when expo-router hands back an array', () => {
    expect(titleForDisplay(['First title', 'Second title'])).toBe('First title');
  });
});
