import type { MissionVM } from '@/services/adapters/missions';
import { nullMetric, type Metric } from '@/services/adapters/provenance';

export function selectMission(missions: MissionVM[], id: string | null | undefined): MissionVM | null {
  if (!id) {
    return null;
  }
  return missions.find((mission) => mission.id === id) ?? null;
}

export interface CanCompleteResult {
  allowed: boolean;
  reason?: string;
}

export function canComplete(mission: MissionVM): CanCompleteResult {
  if (mission.status === 'complete') {
    return { allowed: false, reason: 'This mission is already complete' };
  }

  if (mission.source !== 'life' && mission.source !== 'health') {
    return { allowed: false, reason: 'Completing this mission type arrives with its domain lane' };
  }

  return { allowed: true };
}

export interface MissionTrendPoint {
  x: number | string;
  y: number | null;
  kind?: 'actual' | 'projected' | 'marker';
}

export function progressPoints(mission: MissionVM): MissionTrendPoint[] {
  return [{ x: 'now', y: mission.progress, kind: 'actual' }];
}

export function xpKpi(_mission: MissionVM): Metric<number> {
  return nullMetric<number>('XP confirms after completion');
}

import { describe, expect, it } from 'vitest';

import type { MissionVM } from '@/services/adapters/missions';
import {
  canComplete,
  progressPoints,
  selectMission,
  xpKpi,
} from '@/features/missions/mission-detail-logic';

function buildMission(over: Partial<MissionVM> = {}): MissionVM {
  return {
    id: 'm1',
    title: 'Run 5k',
    description: null,
    tier: 'main',
    status: 'active',
    progress: 40,
    domain: 'fitness',
    source: 'life',
    isPrimary: false,
    dueDate: null,
    target: null,
    ...over,
  };
}

describe('selectMission', () => {
  it('finds a mission by id in a multi-item list', () => {
    const missions = [
      buildMission({ id: 'm1' }),
      buildMission({ id: 'm2', title: 'Read 10 pages' }),
    ];
    expect(selectMission(missions, 'm2')?.title).toBe('Read 10 pages');
  });

  it('returns null for an id not present in a non-empty list', () => {
    const missions = [buildMission({ id: 'm1' })];
    expect(selectMission(missions, 'mX')).toBeNull();
  });

  it('returns null when id is undefined', () => {
    const missions = [buildMission({ id: 'm1' })];
    expect(selectMission(missions, undefined)).toBeNull();
  });

  it('returns null when id is null', () => {
    const missions = [buildMission({ id: 'm1' })];
    expect(selectMission(missions, null)).toBeNull();
  });

  it('returns null for an empty missions array', () => {
    expect(selectMission([], 'm1')).toBeNull();
  });
});

describe('canComplete', () => {
  it('allows completing active life missions', () => {
    expect(canComplete(buildMission({ source: 'life', status: 'active' }))).toEqual({
      allowed: true,
    });
  });

  it('allows completing active health missions', () => {
    expect(canComplete(buildMission({ source: 'health', status: 'active' }))).toEqual({
      allowed: true,
    });
  });

  it('disables career missions with the domain lane reason', () => {
    expect(canComplete(buildMission({ source: 'career', status: 'active' }))).toEqual({
      allowed: false,
      reason: 'Completing this mission type arrives with its domain lane',
    });
  });

  it('disables finance missions with the domain lane reason', () => {
    expect(canComplete(buildMission({ source: 'finance', status: 'active' }))).toEqual({
      allowed: false,
      reason: 'Completing this mission type arrives with its domain lane',
    });
  });

  it('disables already complete life missions', () => {
    expect(canComplete(buildMission({ source: 'life', status: 'complete' }))).toEqual({
      allowed: false,
      reason: 'This mission is already complete',
    });
  });

  it('asserts status precedence over source for complete missions', () => {
    expect(canComplete(buildMission({ source: 'career', status: 'complete' }))).toEqual({
      allowed: false,
      reason: 'This mission is already complete',
    });
  });
});

describe('progressPoints', () => {
  it('returns exactly one point for a mid-range progress value', () => {
    const points = progressPoints(buildMission({ progress: 72 }));
    expect(points).toEqual([{ x: 'now', y: 72, kind: 'actual' }]);
  });

  it('still returns exactly one point when progress is 0', () => {
    const points = progressPoints(buildMission({ progress: 0 }));
    expect(points).toEqual([{ x: 'now', y: 0, kind: 'actual' }]);
  });

  it('returns an array length less than 3 to render as a dot', () => {
    const points = progressPoints(buildMission({ progress: 50 }));
    expect(points.length).toBeLessThan(3);
  });
});

describe('xpKpi', () => {
  it('renders an honest-null metric for active missions', () => {
    const metric = xpKpi(buildMission({ status: 'active' }));
    expect(metric.value).toBeNull();
    expect(metric.provenance.kind).toBe('null');
    expect(metric.provenance.label).toBe('XP confirms after completion');
  });

  it('renders an honest-null metric for already-complete missions', () => {
    const metric = xpKpi(buildMission({ status: 'complete' }));
    expect(metric.value).toBeNull();
    expect(metric.provenance.kind).toBe('null');
  });
});

import React, { useState } from 'react';

const TIER_LABEL: Record<MissionTier, string> = {

  main: 'Main Mission',

  daily: 'Daily Mission',

const SOURCE_LABEL: Record<MissionVM['source'], string> = {
  health: 'Health',
  life: 'Life',
  career: 'Career',
  finance: 'Finance',

function toTitleCase(str: string): string {
  return str.charAt(0).toUpperCase() + str.slice(1);
}

function TierBadge({ tier }: { tier: MissionTier }): JSX.Element {
  const colors = MissionMetalColors[tier];
  return (
    <View
      style={[
        styles.badgeContainer,
        { borderColor: colors.solid, backgroundColor: colors.soft },
      ]}
    >
      <Text style={[styles.badgeText, { color: colors.solid }]}>
        {TIER_LABEL[tier]}
      </Text>
    </View>
  );
}

function ChromeRow({ mission }: { mission: MissionVM | null }): JSX.Element {
  const router = useRouter();
  return (
    <View style={styles.chromeRow}>

function ActionCheckCard({
  mission,
  completing,
  justCompletedNoXp,
  onPress,
}: {
  mission: MissionVM;
  completing: boolean;
  justCompletedNoXp: boolean;
  onPress: () => void;
}): JSX.Element {
  const isDone = mission.status === 'complete';
  const completion = canComplete(mission);
  const glyph = isDone ? '✓' : '○';
  const glyphColor = isDone ? BalenciaColors.green : BalenciaColors.paper100;

  let subcopy = '';
  if (isDone) {
    subcopy = justCompletedNoXp
      ? 'XP confirms within about 15 minutes'
      : 'Mission complete';
  } else {
    subcopy = completion.allowed
      ? 'Tap mark complete when you finish this'
      : completion.reason ?? '';
  }

  return (
    <View
      accessibilityLabel={`${mission.title}, ${isDone ? 'complete' : 'not complete'}`}
      importantForAccessibility="no-hide-descendants"
      style={[
        styles.cardBase,
        styles.actionCard,
        isDone && {
          backgroundColor: BalenciaColors.greenSoft,
          borderColor: 'rgba(52, 168, 83, 0.32)',
        },
      ]}
    >
      <View style={styles.actionRow}>
        <Text style={[styles.glyph, { color: glyphColor }]}>{glyph}</Text>
        <Text style={styles.actionTitle}>{mission.title}</Text>
      </View>
      <Text style={styles.actionSubcopy}>{subcopy}</Text>
      {!isDone && (
        <BalenciaButton
          tone="primary"
          onPress={onPress}
          disabled={!completion.allowed || completing}
          accessibilityLabel="Mark complete"
        >
          {completing ? 'Marking complete...' : 'Mark complete'}
        </BalenciaButton>
      )}
    </View>
  );
}

export function MissionDetailScreen(): JSX.Element {
  const params = useLocalSearchParams<{ id?: string }>();
  const router = useRouter();

  const { data, status, refetch } = useMissions();
  const completeMutation = useCompleteMission();

  const [completeError, setCompleteError] = useState<string | null>(null);
  const [justCompletedNoXp, setJustCompletedNoXp] = useState(false);

  const missions = data?.missions ?? [];
  const mission = selectMission(missions, params.id);

  const handleComplete = () => {
    if (!mission) {
      return;
    }
    if (mission.source !== 'life' && mission.source !== 'health') {
      return;
    }

    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => undefined);
    setCompleteError(null);

    completeMutation.mutate(
      { id: mission.id, source: mission.source },
      {
        onSuccess: (result) => {
          if (result.xpDelta !== null && result.xpDelta > 0) {
            router.push({
              pathname: '/celebration',
              params: { xpDelta: String(result.xpDelta), title: mission.title },
            } as Href);
          } else {
            setJustCompletedNoXp(true);
          }
        },
        onError: (error: Error) => {
          setCompleteError(
            error.message || "Couldn't mark this mission complete. Try again."
          );
        },
      }
    );
  };

  let body: React.ReactNode = null;
  if (status === 'pending') {
    body = <SkeletonState variant="hero" />;
  } else if (status === 'error') {
    body = <ErrorState message="Couldn't load your missions." onRetry={refetch} />;
  } else if (!mission) {
    body = <ErrorState message="This mission isn't available." onRetry={refetch} />;
  } else {
    body = (
      <>
        {/* Hero */}
        <View style={styles.heroCenter}>
          <ProgressRing
            progress={mission.progress}
            size={120}
            stroke={8}
            centerLabel={`${Math.round(mission.progress)}%`}
            glow={mission.status === 'complete' ? 'done' : 'you'}
          />
        </View>
        <Text style={styles.title}>{mission.title}</Text>

        <View style={styles.chipRow}>
          <View style={styles.chipWrapper}>
            <ChipDomainTag label={toTitleCase(mission.domain)} />
          </View>
          <View style={styles.chipWrapper}>
            <ChipDomainTag label={SOURCE_LABEL[mission.source]} />
          </View>
        </View>

        {mission.dueDate ? (
          <Text style={styles.dueDate}>
            {`Due ${mission.dueDate.slice(0, 10)}`}
          </Text>
        ) : null}

        {/* KPI Row */}
        <View style={styles.kpiRow}>
          <View style={styles.kpiItem}>
            <GlassStatCard
              overline="Progress"
              glow="you"
              metric={realMetric(mission.progress, {
                source: 'server',
                label: 'system calculated',
                format: (v) => `${Math.round(v)}%`,
              })}
            />
          </View>
          <View style={styles.kpiItem}>
            <GlassStatCard
              overline="Target"
              glow="you"
              metric={
                mission.target
                  ? realMetric(
                      `${mission.target.current}/${mission.target.target} ${mission.target.unit}`,
                      { source: 'server', label: 'system calculated' }
                    )
                  : nullMetric<string>('No target set for this mission')
              }
            />
          </View>
          <View style={styles.kpiItem}>
            <GlassStatCard overline="XP" glow="you" metric={xpKpi(mission)} />
          </View>
        </View>

        {/* CIA Insight */}
        <View style={styles.sectionSpacing}>
          <CIAInsightCard
            headline={null}
            gated={{ reason: 'Cia reasoning arrives with the coach lane' }}
          />
        </View>

        {/* Error Toast */}
        {completeError ? (
          <View style={styles.sectionSpacing}>
            <ToastBanner
              message={completeError}
              tone="error"
              onDismiss={() => setCompleteError(null)}
            />
          </View>
        ) : null}

        {/* Action Check Card */}
        <View style={styles.sectionSpacing}>
          <ActionCheckCard
            mission={mission}
            completing={completeMutation.isPending}
            justCompletedNoXp={justCompletedNoXp}
            onPress={handleComplete}
          />
        </View>

        {/* Accordions */}
        <View style={styles.sectionSpacing}>
          <ExpandableSection title="All actions">
            <View style={styles.accordionInner}>
              <View style={styles.actionRow}>
                <Text style={styles.glyph}>
                  {mission.status === 'complete' ? '✓' : '○'}
                </Text>
                <Text style={styles.actionTitle}>{mission.title}</Text>
              </View>
              <Text style={styles.caption}>
                Multiple actions arrive with the domain lanes
              </Text>
            </View>
          </ExpandableSection>

          <View style={styles.accordionSpacing} />
          <ExpandableSection title="Progress">
            <View style={styles.accordionInner}>
              <TrendChart
                points={progressPoints(mission)}
                state="real"
                height={120}
              />
              <Text style={styles.caption}>
                One reading available — trend history isn't tracked yet
              </Text>
            </View>
          </ExpandableSection>

          <View style={styles.accordionSpacing} />
          <ExpandableSection title="Milestones">
            <HonestNullState title="No milestones yet" />
          </ExpandableSection>

          <View style={styles.accordionSpacing} />
          <ExpandableSection
            title="Cia reasoning"
            gated={{ reason: 'Cia reasoning arrives with the coach lane' }}
          />

          <View style={styles.accordionSpacing} />
          <ExpandableSection
            title="Cross-domain links"
            gated={{ reason: 'Cross-domain links arrive with the life areas lane' }}
          />
        </View>
      </>
    );
  }

  return (
    <View style={styles.container}>
      <ChromeRow mission={mission} />
      <ScrollView
        contentContainerStyle={styles.scrollContent}
        showsVerticalScrollIndicator={false}
      >
        {body}
      </ScrollView>
    </View>
  );
}
