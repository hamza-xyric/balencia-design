'use client-free';

import React, { useCallback, useMemo } from 'react';
import { StyleSheet, View, ViewStyle, LayoutChangeEvent } from 'react-native';
import Svg, { Path, Circle, Line } from 'react-native-svg';
import Animated, {
  useAnimatedProps,
  useSharedValue,
  withTiming,
  withDelay,
  Easing,
} from 'react-native-reanimated';

import { BalenciaColors, TouchTarget } from '@/constants/theme';
import { MotionDuration, MotionEasing } from '@/constants/motion';
import { useReducedMotion } from '@/components/balencia/auth-inputs';

// Duplicated intentionally from the ProgressRing contract (BIOS-004 §3) to
// preserve this file's strict Wave-0-only dependency profile. Do not refactor
// into a cross-import without checking that the sibling viz file has landed.
export type Honesty = 'real' | 'low' | 'null';

export interface TrendPoint {
  x: number | string;
  y: number | null;
  kind?: 'actual' | 'projected' | 'marker';
}

export interface TrendChartProps {
  points: TrendPoint[];
  height?: number;
  state?: Honesty;
  showProjection?: boolean;
  onScrub?: (p: TrendPoint) => void;
  testID?: string;
}

// Virtual coordinate space
const VIEW_WIDTH = 300;
const VIEW_HEIGHT = 160;
const PADDING = 12;

const AnimatedPath = Animated.createAnimatedComponent(Path);

const gridlines = [PADDING, VIEW_HEIGHT / 2, VIEW_HEIGHT - PADDING];

export function TrendChart({
  points,
  height = 160,
  state = 'real',
  showProjection = false,
  onScrub,
  testID,
}: TrendChartProps) {
  const reduceMotion = useReducedMotion();

  // Touch tracking for scrub
  const measuredWidth = useSharedValue(0);
  const handleLayout = useCallback((e: LayoutChangeEvent) => {
    measuredWidth.value = e.nativeEvent.layout.width;
  }, [measuredWidth]);

  // Separate points by kind
  const { actualPoints, projectedPoints, markerPoints } = useMemo(() => {
    const actual: TrendPoint[] = [];
    const projected: TrendPoint[] = [];
    const marker: TrendPoint[] = [];
    for (const p of points) {
      if (p.y === null) continue;
      const kind = p.kind ?? 'actual';
      if (kind === 'actual') actual.push(p);
      else if (kind === 'projected') projected.push(p);
      else if (kind === 'marker') marker.push(p);
    }
    return { actualPoints: actual, projectedPoints: projected, markerPoints: marker };
  }, [points]);

  // Compute global min/max across all eligible points
  const scale = useMemo(() => {
    const eligible = [...actualPoints, ...projectedPoints, ...markerPoints];
    const validYs = eligible.map((p) => p.y as number);
    if (validYs.length === 0) {
      return { hasData: false, min: 0, max: 0, toX: () => 0, toY: () => 0 };
    }
    const min = Math.min(...validYs);
    const max = Math.max(...validYs);
    const range = max - min || 1;
    const count = eligible.length;

    const toX = (index: number) =>
      count <= 1
        ? PADDING
        : PADDING + (index / (count - 1)) * (VIEW_WIDTH - 2 * PADDING);

    const toY = (val: number) =>
      PADDING +
      (1 - (val - min) / range) * (VIEW_HEIGHT - 2 * PADDING);

    return { hasData: true, min, max, toX, toY };
  }, [actualPoints, projectedPoints, markerPoints]);

  // Map to SVG coordinates using global indices
  const actualCoords = useMemo(() => {
    const arr: { x: number; y: number }[] = [];
    actualPoints.forEach((p, i) => {
      arr.push({ x: scale.toX(i), y: scale.toY(p.y as number) });
    });
    return arr;
  }, [actualPoints, scale]);

  const projectedCoords = useMemo(() => {
    const arr: { x: number; y: number }[] = [];
    // If actuals exist, anchor first projected segment to last actual point
    const startIndex = actualPoints.length > 0 ? actualPoints.length - 1 : 0;
    projectedPoints.forEach((p, idx) => {
      const i = startIndex + (actualPoints.length > 0 ? idx : idx);
      arr.push({ x: scale.toX(i), y: scale.toY(p.y as number) });
    });
    return arr;
  }, [projectedPoints, actualPoints.length, scale]);

  const markerCoords = useMemo(() => {
    const arr: { x: number; y: number }[] = [];
    markerPoints.forEach((p) => {
      const i = [...actualPoints, ...projectedPoints].length;
      arr.push({ x: scale.toX(i), y: scale.toY(p.y as number) });
    });
    return arr;
  }, [markerPoints, scale, actualPoints, projectedPoints]);

  // Calculate exact length of straight-line path
  const actualPathLength = useMemo(() => {
    if (actualCoords.length < 2) return 0;
    let len = 0;
    for (let i = 1; i < actualCoords.length; i++) {
      const dx = actualCoords[i].x - actualCoords[i - 1].x;
      const dy = actualCoords[i].y - actualCoords[i - 1].y;
      len += Math.sqrt(dx * dx + dy * dy);
    }
    return len;
  }, [actualCoords]);

  // Animation shared values
  const dashOffset = useSharedValue(reduceMotion ? 0 : actualPathLength);
  const projectedOpacity = useSharedValue(
    reduceMotion || actualPoints.length === 0 ? 1 : 0
  );

  React.useEffect(() => {
    if (reduceMotion || state === 'null') {
      dashOffset.value = 0;
      projectedOpacity.value = 1;
      return;
    }
    dashOffset.value = withTiming(0, {
      duration: MotionDuration.draw,
      easing: Easing.bezier(...MotionEasing.physical),
    });
    projectedOpacity.value = withDelay(
      MotionDuration.draw,
      withTiming(1, {
        duration: MotionDuration.draw,
        easing: Easing.bezier(...MotionEasing.physical),
      })
    );
  }, [reduceMotion, state, dashOffset, projectedOpacity]);

  const animatedActualProps = useAnimatedProps(() => ({
    strokeDashoffset: dashOffset.value,
  }));

  const animatedProjectedProps = useAnimatedProps(() => ({
    opacity: projectedOpacity.value,
  }));

  // Render gates based on Honesty state
  const showNull = state === 'null';
  const showLow = state === 'low';
  const hasLine = actualCoords.length >= 3;
  const showProjectionLine = showProjection && projectedCoords.length >= 1;

  // Touch responder for scrubbing
  const handleScrub = useCallback(
    (clientX: number) => {
      const width = measuredWidth.value;
      if (width === 0) return;
      const ratio = Math.min(1, Math.max(0, clientX / width));
      const eligible = [...actualPoints, ...projectedPoints];
      if (eligible.length === 0) return;
      const index = Math.round(ratio * (eligible.length - 1));
      const point = eligible[Math.min(eligible.length - 1, Math.max(0, index))];
      if (point) {
        onScrub?.(point);
      }
    },
    [measuredWidth, actualPoints, projectedPoints, onScrub]
  );

  const touchHandlers = useMemo(
    () =>
      onScrub && !showNull
        ? {
            onStartShouldSetResponder: () => true,
            onResponderGrant: (e: { nativeEvent: { locationX: number } }) =>
              handleScrub(e.nativeEvent.locationX),
            onResponderMove: (e: { nativeEvent: { locationX: number } }) =>
              handleScrub(e.nativeEvent.locationX),
          }
        : {},
    [onScrub, showNull, handleScrub]
  );

  // hitSlop for 44px minimum touch target
  const hitSlop = useMemo(() => {
    if (!onScrub || height >= TouchTarget) return undefined;
    const pad = Math.ceil((TouchTarget - height) / 2);
    return { top: pad, bottom: pad, left: 0, right: 0 };
  }, [onScrub, height]);

  const containerStyle: ViewStyle = {
    height,
    opacity: showNull ? 0.5 : 1,
  };

  // Format SVG Path "d" string
  const getD = (coords: { x: number; y: number }[]) => {
    if (coords.length === 0) return '';
    let d = `M ${coords[0].x} ${coords[0].y}`;
    for (let i = 1; i < coords.length; i++) {
      d += ` L ${coords[i].x} ${coords[i].y}`;
    }
    return d;
  };

  return (
    <View
      testID={testID}
      accessible
      accessibilityRole="image"
      accessibilityLabel="Trend chart"
      style={containerStyle}
      onLayout={handleLayout}
      hitSlop={hitSlop}
      {...touchHandlers}
    >
      <Svg
        width="100%"
        height="100%"
        viewBox={`0 0 ${VIEW_WIDTH} ${VIEW_HEIGHT}`}
        preserveAspectRatio="none"
      >
        {/* 5. Gridlines */}
        {gridlines.map((y, i) => (
          <Line
            key={`grid-${i}`}
            x1={PADDING}
            y1={y}
            x2={VIEW_WIDTH - PADDING}
            y2={y}
            stroke={BalenciaColors.chartGridline}
            strokeWidth={1}
          />
        ))}

        {/* Suppress all geometry in 'null' state */}
        {!showNull && scale.hasData && (
          <>
            {/* 7. Marker dots (independent of line threshold) */}
            {markerCoords.map((c, i) => (
              <Circle
                key={`marker-${i}`}
                cx={c.x}
                cy={c.y}
                r={3}
                fill={BalenciaColors.green}
              />
            ))}

            {/* 2. Line vs. Dots threshold */}
            {hasLine ? (
              <AnimatedPath
                d={getD(actualCoords)}
                fill="none"
                stroke={BalenciaColors.orange}
                strokeWidth={2}
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeDasharray={`${actualPathLength} ${actualPathLength}`}
                animatedProps={animatedActualProps}
                opacity={showLow ? 0.4 : 1}
              />
            ) : (
              actualCoords.map((c, i) => (
                <Circle
                  key={`actual-dot-${i}`}
                  cx={c.x}
                  cy={c.y}
                  r={3}
                  fill={BalenciaColors.orange}
                  opacity={showLow ? 0.4 : 1}
                />
              ))
            )}

            {/* 4. Projection line (opt-in via showProjection) */}
            {showProjectionLine && (
              <AnimatedPath
                d={getD(projectedCoords)}
                fill="none"
                stroke={BalenciaColors.purple}
                strokeWidth={2}
                strokeLinecap="round"
                strokeLinejoin="round"
                strokeDasharray="6 6"
                animatedProps={animatedProjectedProps}
              />
            )}
          </>
        )}
      </Svg>
    </View>
  );
}
