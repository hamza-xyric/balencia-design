import { apiFetch } from '@/services/api/client';
import type {
  AiCoachChatResponseDto,
} from '@/services/api/dto/ai-coach';
import type {
  CiaMessage,
  FitnessDashboard,
  GamificationStats,
  LifeArea,
  LoginRequest,
  LoginResponse,
  Mission,
  UserSummary,
} from '@/types/pilot';

type UnifiedGoalsPayload = {
  goals?: Mission[];
  missions?: Mission[];
};

type LifeAreasPayload = {
  areas?: LifeArea[];
  domains?: LifeArea[];
};

type WorkoutPlansPayload = {
  plans?: unknown[];
};

export function login(payload: LoginRequest) {
  return apiFetch<LoginResponse>('/auth/login', {
    method: 'POST',
    skipAuth: true,
    body: payload,
  });
}

export function getCurrentUser() {
  return apiFetch<{ user?: UserSummary } | UserSummary>('/auth/me');
}

export async function getUnifiedMissions() {
  const payload = await apiFetch<UnifiedGoalsPayload>('/v1/goals/unified');
  return payload.missions ?? payload.goals ?? [];
}

export async function getLifeAreas() {
  const payload = await apiFetch<LifeAreasPayload>('/life-areas/summary');
  return payload.areas ?? payload.domains ?? [];
}

export async function getGamificationStats() {
  const payload = await apiFetch<{ stats?: GamificationStats }>('/gamification/stats');
  return payload.stats;
}

export async function getFitnessDashboard(): Promise<FitnessDashboard | null> {
  const payload = await apiFetch<WorkoutPlansPayload>('/workouts/plans');
  if (!payload.plans) {
    return null;
  }

  return {
    score: 74,
    weeklyMinutes: payload.plans.length * 38,
    recovery: 68,
    trend: [32, 44, 39, 62, 70, 58, 76],
    metrics: [
      { label: 'Plans', value: String(payload.plans.length), detail: 'Workout plans from backend' },
      { label: 'Readiness', value: '68%', detail: 'Derived until native fitness contract is finalized' },
      { label: 'Load', value: 'Moderate', detail: 'Flagged mobile transformation' },
    ],
  };
}

export type SendCiaChatArgs = {
  message: string;
  sessionId?: string;
  isOnboarding?: boolean;
  goal?: string;
};

export function buildCiaRequest(args: SendCiaChatArgs) {
  return {
    method: 'POST',
    body: {
      message: args.message,
      sessionId: args.sessionId,
      goal: args.goal ?? 'overall_optimization',
      isOnboarding: args.isOnboarding ?? false,
    },
    headers: {
      'Idempotency-Key': `cia-${Date.now()}-${Math.random().toString(36).slice(2)}`,
    },
  };
}

export async function sendCiaChat(args: SendCiaChatArgs): Promise<AiCoachChatResponseDto> {
  return apiFetch<AiCoachChatResponseDto>(
    '/ai-coach/chat',
    buildCiaRequest(args),
  );
}

import { ApiError } from '@/services/api/errors';
import type { AiCoachChatResponseDto } from '@/services/api/dto/ai-coach';

export type CiaGate =
  | 'feature_disabled'
  | 'credits_exhausted'
  | 'plan_upgrade_required'
  | 'client_error'
  | null;

export type CiaMessageVM = {
  id: string;
  role: 'user' | 'cia';
  text: string;
  phase?: string;
  suggestedActions?: string[];
  pending?: boolean;
  failed?: boolean;
};

export function toCiaMessage(dto: AiCoachChatResponseDto): CiaMessageVM {
  return {
    id: `cia-${dto.sessionId}-${Date.now()}`,
    role: 'cia',
    text: dto.message,
    phase: dto.phase,
    suggestedActions: dto.suggestedActions,
  };
}

export function classifyCiaError(err: unknown): { gate: CiaGate; userMessage: string } {
  if (err instanceof ApiError) {
    if (err.status === 402 && err.code === 'FEATURE_DISABLED') {
      return {
        gate: 'feature_disabled',
        userMessage: "Cia coaching isn't part of your current plan.",
      };
    }

    if (err.status === 402 && err.code === 'CREDITS_EXHAUSTED') {
      return {
        gate: 'credits_exhausted',
        userMessage: "You've used all your coaching credits for now.",
      };
    }

    if (err.status === 400 && err.code === 'MISSING_IDEMPOTENCY_KEY') {
      return {
        gate: 'client_error',
        userMessage: 'Something went wrong sending that. Try again.',
      };
    }

    if (err.status === 402 && err.code === 'PLAN_UPGRADE_REQUIRED') {
      return {
        gate: 'plan_upgrade_required',
        userMessage: 'This needs a plan upgrade.',
      };
    }
  }

  return {
    gate: null,
    userMessage: "Cia couldn't reply just now. Tap to retry.",
  };
}

import { useReducer, useCallback } from 'react';
import {
  sendCiaChat,
} from '@/services/api/pilot';
import {
  classifyCiaError,
  toCiaMessage,
  type CiaGate,
  type CiaMessageVM,
} from '@/services/adapters/cia';

export type CiaChatState = {
  messages: CiaMessageVM[];
  sessionId: string | null;
  sending: boolean;
  gate: CiaGate;
  gateMessage: string | null;
};

type Action =
  | { type: 'ADD_USER_MESSAGE'; message: CiaMessageVM }
  | { type: 'ADD_PENDING_CIA'; message: CiaMessageVM }
  | { type: 'REPLACE_PENDING_CIA'; pendingId: string; message: CiaMessageVM; sessionId: string }
  | { type: 'MARK_FAILED'; userId: string; gate: CiaGate; gateMessage: string }
  | { type: 'CLEAR_FAILED'; userId: string }
  | { type: 'RESET_GATE' };

const initialState: CiaChatState = {
  messages: [],
  sessionId: null,
  sending: false,
  gate: null,
  gateMessage: null,
};

function reducer(state: CiaChatState, action: Action): CiaChatState {
  switch (action.type) {
    case 'ADD_USER_MESSAGE': {
      return {
        ...state,
        gate: null,
        gateMessage: null,
        sending: true,
        messages: [...state.messages, action.message],
      };
    }
    case 'ADD_PENDING_CIA': {
      return {
        ...state,
        messages: [...state.messages, action.message],
      };
    }
    case 'REPLACE_PENDING_CIA': {
      const filtered = state.messages.filter(m => m.id !== action.pendingId);
      return {
        ...state,
        sending: false,
        sessionId: action.sessionId,
        gate: null,
        gateMessage: null,
        messages: [...filtered, action.message],
      };
    }
    case 'MARK_FAILED': {
      const updated = state.messages.map(m =>
        m.id === action.userId ? { ...m, failed: true } : m
      ).filter(m => !m.pending);
      
      return {
        ...state,
        sending: false,
        messages: updated,
        gate: action.gate,
        gateMessage: action.gateMessage,
      };
    }
    case 'CLEAR_FAILED': {
      return {
        ...state,
        gate: null,
        gateMessage: null,
        messages: state.messages.map(m =>
          m.id === action.userId ? { ...m, failed: false } : m
        ),
      };
    }
    case 'RESET_GATE': {
      return {
        ...state,
        gate: null,
        gateMessage: null,
      };
    }
    default:
      return state;
  }
}

export function useCiaChat(opts?: {
  isOnboarding?: boolean;
  starter?: CiaMessageVM;
}): CiaChatState & {
  send(text: string): Promise<void>;
  retry(messageId: string): Promise<void>;
} {
  const initial: CiaChatState = opts?.starter
    ? { ...initialState, messages: [opts.starter] }
    : initialState;

  const [state, dispatch] = useReducer(reducer, initial);

  const performSend = useCallback(async (text: string, userId: string) => {
    const pendingId = `cia-pending-${Date.now()}`;

    dispatch({
      type: 'ADD_PENDING_CIA',
      message: {
        id: pendingId,
        role: 'cia',
        text: '',
        pending: true,
      },
    });

    try {
      const response = await sendCiaChat({
        message: text,
        sessionId: state.sessionId ?? undefined,
        isOnboarding: opts?.isOnboarding,
      });

      const ciaMessage = toCiaMessage(response);

      dispatch({
        type: 'REPLACE_PENDING_CIA',
        pendingId,
        message: ciaMessage,
        sessionId: response.sessionId,
      });
    } catch (err) {
      const { gate, userMessage } = classifyCiaError(err);
      dispatch({
        type: 'MARK_FAILED',
        userId,
        gate,
        gateMessage: userMessage,
      });
    }
  }, [opts?.isOnboarding, state.sessionId]);

  const send = useCallback(async (text: string) => {
    const userId = `user-${Date.now()}`;
    dispatch({
      type: 'ADD_USER_MESSAGE',
      message: {
        id: userId,
        role: 'user',
        text,
      },
    });

    await performSend(text, userId);
  }, [performSend]);

  const retry = useCallback(async (messageId: string) => {
    const message = state.messages.find(m => m.id === messageId);
    if (!message) {
      return;
    }

    dispatch({ type: 'CLEAR_FAILED', userId: messageId });
    await performSend(message.text, messageId);
  }, [performSend, state.messages]);

  return {
    messages: state.messages,
    sessionId: state.sessionId,
    sending: state.sending,
    gate: state.gate,
    gateMessage: state.gateMessage,
    send,
    retry,
  };
}
