export interface ChipDomainTagProps {
  label: string;
  active?: boolean;
  onPress?: () => void;
  testID?: string;
}

export function ChipDomainTag(props: ChipDomainTagProps): JSX.Element {
  const { label, active, onPress, testID } = props;

  if (onPress) {
    return (
      <Pressable
        onPress={onPress}
        hitSlop={{ top: 6, bottom: 6, left: 6, right: 6 }}
        accessibilityRole="button"
        accessibilityState={{ selected: !!active }}
        accessibilityLabel={label}
        testID={testID}
        style={({ pressed }) => [
          styles.chip,
          active ? styles.chipActive : styles.chipInactive,
          {
            opacity: pressed ? 0.86 : 1,
            transform: [{ scale: pressed ? 0.98 : 1 }],
          },
        ]}
      >
        <Text style={[styles.chipLabel, active ? styles.chipLabelActive : styles.chipLabelInactive]}>
          {label}
        </Text>
      </Pressable>
    );
  }

  return (
    <View
      style={[styles.chip, active ? styles.chipActive : styles.chipInactive]}
      accessible
      accessibilityRole="text"
      accessibilityLabel={label}
      testID={testID}
    >
      <Text style={[styles.chipLabel, active ? styles.chipLabelActive : styles.chipLabelInactive]}>
        {label}
      </Text>
    </View>
  );
}

export interface SegmentedTabsOption<T extends string = string> {
  key: T;
  label: string;
}

export interface SegmentedTabsProps<T extends string = string> {
  options: SegmentedTabsOption<T>[];
  value: T;
  onChange: (key: T) => void;
  accessibilityLabel?: string;
  testID?: string;
}

function SegmentedTabsSegment<T extends string>({
  option,
  isActive,
  reduceMotion,
  onSelect,
}: {
  option: SegmentedTabsOption<T>;
  isActive: boolean;
  reduceMotion: boolean;
  onSelect: (key: T) => void;
}) {
  const scale = useSharedValue(1);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  const handlePressIn = () => {
    if (!reduceMotion) {
      scale.value = withTiming(0.96, {
        duration: MotionDuration.feedback,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }
  };

  const handlePressOut = () => {
    if (!reduceMotion) {
      scale.value = withTiming(1, {
        duration: MotionDuration.feedback,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }
  };

  return (
    <Pressable
      onPress={() => onSelect(option.key)}
      onPressIn={handlePressIn}
      onPressOut={handlePressOut}
      accessibilityRole="tab"
      accessibilityState={{ selected: isActive }}
      accessibilityLabel={option.label}
      style={styles.segment}
    >
      <Animated.View
        style={[
          styles.segmentInner,
          isActive ? styles.segmentActive : styles.segmentInactive,
          animatedStyle,
        ]}
      >
        <Text
          style={[
            styles.segmentLabel,
            isActive ? styles.segmentLabelActive : styles.segmentLabelInactive,
          ]}
        >
          {option.label}
        </Text>
      </Animated.View>
    </Pressable>
  );
}

export function SegmentedTabs<T extends string = string>(props: SegmentedTabsProps<T>): JSX.Element {
  const { options, value, onChange, accessibilityLabel, testID } = props;
  const reduceMotion = useReducedMotion();

  const handleSelect = (key: T) => {
    if (key !== value) {
      onChange(key);
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => undefined);
    }
  };

  return (
    <View
      style={styles.tabsTrack}
      accessibilityRole="tablist"
      accessibilityLabel={accessibilityLabel}
      testID={testID}
    >
      {options.map((option) => (
        <SegmentedTabsSegment
          key={option.key}
          option={option}
          isActive={option.key === value}
          reduceMotion={reduceMotion}
          onSelect={handleSelect}
        />
      ))}
    </View>
  );
}

export interface FABQuickLogProps {
  onPress: () => void;
  accessibilityLabel: string;
  testID?: string;
}

export function FABQuickLog(props: FABQuickLogProps): JSX.Element {
  const { onPress, accessibilityLabel, testID } = props;
  const insets = useSafeAreaInsets();
  const reduceMotion = useReducedMotion();

  const scale = useSharedValue(1);

  const animatedStyle = useAnimatedStyle(() => ({
    transform: [{ scale: scale.value }],
  }));

  const handlePressIn = () => {
    if (!reduceMotion) {
      scale.value = withTiming(0.96, {
        duration: MotionDuration.feedback,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }
  };

  const handlePressOut = () => {
    if (!reduceMotion) {
      scale.value = withTiming(1, {
        duration: MotionDuration.feedback,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }
  };

  const handlePress = () => {
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => undefined);
    onPress();
  };

  return (
    <Pressable
      onPress={handlePress}
      onPressIn={handlePressIn}
      onPressOut={handlePressOut}
      accessibilityRole="button"
      accessibilityLabel={accessibilityLabel}
      testID={testID}
      style={[
        styles.fab,
        {
          bottom: insets.bottom + BottomTabInset + Spacing.four,
        },
      ]}
    >
      <Animated.View style={[styles.fabCircle, animatedStyle]}>
        <SymbolView
          name={{ ios: 'plus', android: 'add', web: 'add' }}
          size={24}
          weight="bold"
          tintColor={BalenciaColors.paper100}
        />
      </Animated.View>
    </Pressable>
  );
}

export interface FrostCardProps {
  glow?: 'done' | 'cia' | 'you';
  breathe?: boolean;
  breatheConfig?: { durationMs: number; opacityMin: number; opacityMax: number };
  children: ReactNode;
  testID?: string;
}

export function FrostCard(props: FrostCardProps): JSX.Element {
  const { glow, breathe = false, breatheConfig, children, testID } = props;
  const reduceMotion = useReducedMotion();

  const config = breatheConfig ?? {
    durationMs: MotionBreathe.durationMs,
    opacityMin: MotionBreathe.opacityMin,
    opacityMax: MotionBreathe.opacityMax,
  };

  const breatheOpacity = useSharedValue(config.opacityMax);

  useEffect(() => {
    if (!glow) {
      return;
    }

    if (reduceMotion || !breathe) {
      breatheOpacity.value = config.opacityMax;
      return;
    }

    breatheOpacity.value = config.opacityMin;
    breatheOpacity.value = withRepeat(
      withTiming(config.opacityMax, {
        duration: config.durationMs,
        easing: Easing.inOut(Easing.ease),
      }),
      -1,
      true,
    );
  }, [glow, breathe, reduceMotion, breatheOpacity, config.durationMs, config.opacityMin, config.opacityMax]);

  const glowAnimatedStyle = useAnimatedStyle(() => ({
    opacity: breatheOpacity.value,
  }));

  const glowStyle = glow
    ? glow === 'you'
      ? styles.glowYou
      : glow === 'done'
        ? styles.glowDone
        : styles.glowCia
    : null;

  return (
    <View style={styles.frostCard} testID={testID}>
      {glow ? (
        <Animated.View
          style={[styles.glowOverlay, glowStyle, glowAnimatedStyle]}
          pointerEvents="none"
        />
      ) : null}
      {children}
    </View>
  );
}

export interface CIAInsightCardProps {
  headline: string | null;
  body?: string | null;
  gated?: { reason: string } | null;
  testID?: string;
}

export function CIAInsightCard(props: CIAInsightCardProps): JSX.Element {
  const { headline, body, gated, testID } = props;

  let content: ReactNode;
  let accessibilityLabel: string;

  if (gated) {
    content = (
      <View style={styles.ciaGatedRow}>
        <SymbolView
          name={{ ios: 'lock.fill', android: 'lock', web: 'lock' }}
          size={16}
          tintColor={BalenciaColors.purpleText}
        />
        <Text style={styles.ciaBodyText}>{gated.reason}</Text>
      </View>
    );
    accessibilityLabel = `Cia insight, locked: ${gated.reason}`;
  } else if (headline === null) {
    content = <Text style={styles.ciaBodyText}>{HONEST_NULL_TEXT}</Text>;
    accessibilityLabel = `Cia insight: ${HONEST_NULL_TEXT}`;
  } else {
    content = (
      <View>
        <Text style={styles.ciaHeadlineText}>{headline}</Text>
        {body ? <Text style={[styles.ciaBodyText, { marginTop: Spacing.one }]}>{body}</Text> : null}
      </View>
    );
    accessibilityLabel = `Cia insight: ${headline}${body ? `. ${body}` : ''}`;
  }

  return (
    <FrostCard glow="cia" breathe={false} testID={testID}>
      <View style={styles.ciaHeader}>
        <CIAPresenceOrb state="idle" size={28} />
        <Text style={styles.ciaOverline}>Cia insight</Text>
      </View>
      <View
        style={styles.ciaContent}
        accessibilityRole="summary"
        accessibilityLabel={accessibilityLabel}
      >
        {content}
      </View>
    </FrostCard>
  );
}

export interface ExpandableSectionProps {
  title: string;
  subtitle?: string;
  initiallyOpen?: boolean;
  gated?: { reason: string } | null;
  children?: ReactNode;
  testID?: string;
}

export function ExpandableSection(props: ExpandableSectionProps): JSX.Element {
  const { title, subtitle, initiallyOpen = false, gated, children, testID } = props;
  const reduceMotion = useReducedMotion();

  const [isOpen, setIsOpen] = useState(gated ? false : initiallyOpen);
  const [measuredHeight, setMeasuredHeight] = useState(0);

  const openValue = useSharedValue(isOpen ? 1 : 0);

  const toggle = () => {
    if (gated) return;

    setIsOpen((prev) => {
      const next = !prev;
      openValue.value = reduceMotion
        ? next
          ? 1
          : 0
        : withTiming(next ? 1 : 0, {
            duration: MotionDuration.accordion,
            easing: Easing.bezier(...MotionEasing.physical),
          });
      return next;
    });
  };

  const handleLayout = (e: LayoutChangeEvent) => {
    setMeasuredHeight(e.nativeEvent.layout.height);
  };

  const paneAnimatedStyle = useAnimatedStyle(() => ({
    height: measuredHeight === 0 ? undefined : openValue.value * measuredHeight,
    opacity: openValue.value,
  }));

  const chevronAnimatedStyle = useAnimatedStyle(() => ({
    transform: [{ rotate: `${openValue.value * 180}deg` }],
  }));

  const accessibilityLabel = `${title}${gated ? `, locked: ${gated.reason}` : isOpen ? ', expanded' : ', collapsed'}`;

  return (
    <View testID={testID}>
      <Pressable
        onPress={toggle}
        disabled={!!gated}
        accessibilityRole="button"
        accessibilityState={{ expanded: isOpen, disabled: !!gated }}
        accessibilityLabel={accessibilityLabel}
        style={styles.sectionHeader}
      >
        <View>
          <Text style={styles.sectionTitle}>{title}</Text>
          {subtitle ? <Text style={styles.sectionSubtitle}>{subtitle}</Text> : null}
        </View>
        {gated ? (
          <SymbolView
            name={{ ios: 'lock.fill', android: 'lock', web: 'lock' }}
            size={18}
            tintColor={BalenciaColors.paper55}
          />
        ) : (
          <Animated.View style={chevronAnimatedStyle}>
            <SymbolView
              name={{ ios: 'chevron.down', android: 'expand_more', web: 'expand_more' }}
              size={18}
              tintColor={BalenciaColors.paper70}
            />
          </Animated.View>
        )}
      </Pressable>

      <Animated.View style={[styles.sectionPane, paneAnimatedStyle]} pointerEvents={isOpen ? 'auto' : 'none'}>
        <View onLayout={handleLayout}>
          {gated ? (
            <Text style={styles.gatedReasonText}>{gated.reason}</Text>
          ) : (
            children
          )}
        </View>
      </Animated.View>
    </View>
  );
}
