import type { Provenance } from '@/components/balencia/states';

export type Metric<T = number> = {
  value: T | null;
  display: string;
  provenance: Provenance;
};

type RealSource = Extract<Provenance, { kind: 'real' }>['source'];

export function realMetric<T>(
  value: T,
  opts: {
    source: RealSource;
    label?: string;
    format?: (v: T) => string;
  },
): Metric<T> {
  const format = opts.format ?? ((v: T) => String(v));
  const label = opts.label ?? 'Real data';
  return {
    value,
    display: format(value),
    provenance: { kind: 'real', source: opts.source, label },
  };
}

export function nullMetric<T = number>(label?: string): Metric<T> {
  return {
    value: null,
    display: '—',
    provenance: { kind: 'null', label: label ?? 'Not enough data yet' },
  };
}

export function lowMetric<T>(
  value: T,
  label: string,
  format?: (v: T) => string,
): Metric<T> {
  const fmt = format ?? ((v: T) => String(v));
  return {
    value,
    display: fmt(value),
    provenance: { kind: 'low', label },
  };
}

export function fromNullable<T>(
  value: T | null | undefined,
  opts: {
    source: RealSource;
    label?: string;
    nullLabel?: string;
    format?: (v: T) => string;
  },
): Metric<T> {
  if (value === null || value === undefined) {
    return nullMetric<T>(opts.nullLabel);
  }
  return realMetric(value, {
    source: opts.source,
    label: opts.label,
    format: opts.format,
  });
}

  whoop: 'via WHOOP',
  self_report: 'you logged',
  computed: 'computed',
};

function mapPulseMetric(metric: OverviewPulseMetric, hasWearable: boolean): Metric<number> {
  // Enforce the wearable requirement for physiological metrics
  if (!hasWearable && (metric.key === 'recovery' || metric.key === 'stress')) {
    return nullMetric('Connect a wearable');
  }

  // Respect server null values (Honesty Invariant)
  if (metric.value === null || metric.source === null) {
    return nullMetric();
  }

  return realMetric(metric.value, {
    source: metric.source,
    label: pulseMetricLabels[metric.source],
    format: () => metric.display ?? String(metric.value),
  });
}

export function toTodayView(dto: OverviewDashboardResponse): TodayVM {
  const { lifeScore, pulse, dailyBrief, priorities, momentum, domains, meta } = dto;

  const pulseMetrics = (pulse?.metrics ?? []).map((m) => ({
    key: m.key as 'energy' | 'recovery' | 'stress',
    metric: mapPulseMetric(m, pulse?.hasWearable ?? false),
    caption: m.caption ?? null,
    qualityLabel: m.qualityLabel ?? null,
  }));

  const vmPriorities = (priorities ?? []).map((p) => ({
    id: p.id,
    action: p.action,
    pillar: p.pillar ?? 'general',
    priority: (p.priority ?? 'medium') as 'high' | 'medium' | 'low',
  }));

  const vmDomains = (domains ?? []).map((d) => ({
    key: d.id,
    label: d.label,
    score: fromNullable(d.score, {
      source: 'server',
      label: 'computed',
      nullLabel: 'No data yet — start tracking',
    }),
    status: (d.status ?? 'no_data') as 'strong' | 'steady' | 'attention' | 'no_data',
  }));

  return {
    lifePower: fromNullable(lifeScore?.totalScore, {
      source: 'computed',
      label: 'computed',
    }),
    pulse: {
      score: fromNullable(pulse?.score, { source: 'computed', label: 'computed' }),
      metrics: pulseMetrics,
      tip: pulse?.tip ?? null,
      hasWearable: pulse?.hasWearable ?? false,

import type { UnifiedGoalsResponseDto, UnifiedGoalDto, GoalSource } from '@/services/api/dto/goals';

export type MissionTier = 'life' | 'main' | 'side';

export type MissionVM = {
  id: string;
  title: string;
  description: string | null;
  tier: MissionTier;
  status: 'active' | 'blocked' | 'complete';
  progress: number;
  domain: string;
  source: GoalSource;
  isPrimary: boolean;
  dueDate: string | null;
  target: { current: number; target: number; unit: string } | null;
};

export type MissionsVM = {
  missions: MissionVM[];
  summary: {
    total: number;
    active: number;
    completed: number;
    avgProgress: number;
  };
};

function mapTier(goal: UnifiedGoalDto): MissionTier {
  if (goal.isPrimary && goal.source === 'life') {
    return 'life';
  }
  if (goal.isPrimary) {
    return 'main';
  }
  return 'side';
}

function mapStatus(goal: UnifiedGoalDto): MissionVM['status'] {
  switch (goal.status) {
    case 'active':
    case 'in_progress':
      return 'active';
    case 'paused':
      return 'blocked';
    case 'completed':
      return 'complete';
    default:
      return 'active';
  }
}

export function toMissionsView(dto: UnifiedGoalsResponseDto): MissionsVM {
  const goals = dto.goals ?? [];
  
  const missions: MissionVM[] = goals.map((goal) => {
    return {
      id: goal.id,
      title: goal.title,
      description: goal.description ?? null,
      tier: mapTier(goal),
      status: mapStatus(goal),
      progress: goal.progress ?? 0,
      domain: goal.domain ?? 'general',
      source: goal.source,
      isPrimary: goal.isPrimary ?? false,
      dueDate: goal.dueDate ?? null,
      target: goal.target ?? null,
    };
  });

  const summary = dto.summary ?? {
    total: 0,
    active: 0,
    inProgress: 0,
    paused: 0,
    completed: 0,
    avgProgress: 0,
    bySource: {},
  };

  return {
    missions,
    summary: {
      total: summary.total ?? 0,
      active: (summary.active ?? 0) + (summary.inProgress ?? 0),
      completed: summary.completed ?? 0,
      avgProgress: summary.avgProgress ?? 0,
    },
  };
}

    role: 'cia',
    text: dto.message,

    suggestedActions: dto.suggestedActions,

export function classifyCiaError(err: unknown): { gate: CiaGate; userMessage: string } {
  if (err instanceof ApiError) {
    if (err.status === 402 && err.code === 'FEATURE_DISABLED') {
      return {
        gate: 'feature_disabled',
        userMessage: "Cia coaching isn't part of your current plan.",
      };
    }

    if (err.status === 402 && err.code === 'CREDITS_EXHAUSTED') {
      return {
        gate: 'credits_exhausted',
        userMessage: "You've used all your coaching credits for now.",
      };
    }

    if (err.status === 400 && err.code === 'MISSING_IDEMPOTENCY_KEY') {
      return {
        gate: 'client_error',
        userMessage: 'Something went wrong sending that. Try again.',
      };
    }

    if (err.status === 402 && err.code === 'PLAN_UPGRADE_REQUIRED') {
      return {
        gate: 'plan_upgrade_required',
        userMessage: 'This needs a plan upgrade.',
      };
    }
  }

  return {
    gate: null,
    userMessage: "Cia couldn't reply just now. Tap to retry.",
  };
}

import type { PublicUserProfileDto } from '@/services/api/dto/auth';

export type SessionStatus =
  | 'booting'
  | 'unauthenticated'
  | 'authenticating'
  | 'authenticated'
  | 'refreshing'
  | 'expired';

export type SessionState = {
  status: SessionStatus;
  user: PublicUserProfileDto | null;
  error: string | null;
};

export type SessionEvent =
  | { type: 'HYDRATED'; user: PublicUserProfileDto }
  | { type: 'NO_SESSION' }
  | { type: 'SIGN_IN_START' }
  | { type: 'SIGN_IN_SUCCESS'; user: PublicUserProfileDto }
  | { type: 'SIGN_IN_FAILURE'; error: string }
  | { type: 'REFRESH_START' }
  | { type: 'REFRESH_SUCCESS' }
  | { type: 'REFRESH_FAILURE' }
  | { type: 'USER_UPDATED'; user: PublicUserProfileDto }
  | { type: 'SIGNED_OUT' };

export const initialSessionState: SessionState = {
  status: 'booting',
  user: null,
  error: null,
};

export function sessionReducer(state: SessionState, event: SessionEvent): SessionState {
  switch (event.type) {
    case 'HYDRATED': {
      return { status: 'authenticated', user: event.user, error: null };
    }
    case 'NO_SESSION': {
      return { status: 'unauthenticated', user: null, error: null };
    }
    case 'SIGN_IN_START': {
      if (state.status === 'authenticated' || state.status === 'authenticating') {
        return state;
      }
      return { status: 'authenticating', user: null, error: null };
    }
    case 'SIGN_IN_SUCCESS': {
      if (state.status !== 'authenticating') {
        return state;
      }
      return { status: 'authenticated', user: event.user, error: null };
    }
    case 'SIGN_IN_FAILURE': {
      if (state.status !== 'authenticating') {
        return state;
      }
      return { status: 'unauthenticated', user: null, error: event.error };
    }
    case 'REFRESH_START': {
      if (state.status !== 'authenticated') {
        return state;
      }
      return { ...state, status: 'refreshing', error: null };
    }
    case 'REFRESH_SUCCESS': {
      if (state.status !== 'refreshing') {
        return state;
      }
      return { ...state, status: 'authenticated', error: null };
    }
    case 'REFRESH_FAILURE': {
      return { status: 'expired', user: state.user, error: 'session-expired' };
    }
    case 'USER_UPDATED': {
      if (state.status !== 'authenticated' && state.status !== 'refreshing') {
        return state;
      }
      return { ...state, user: event.user };
    }
    case 'SIGNED_OUT': {
      return { status: 'unauthenticated', user: null, error: null };
    }
    default: {
      return state;
    }
  }
}
