import { StyleSheet, Text, View } from 'react-native';
import type { JSX } from 'react';

import type { Metric } from '@/services/adapters/provenance';
import { ProvenanceChip } from '@/components/balencia/states';
import { useReducedMotion } from '@/components/balencia/auth-inputs';
import { ProgressRing } from '@/components/balencia/progress-ring';
import { TrendChart } from '@/components/balencia/trend-chart';
import { BalenciaColors, Radius, Spacing } from '@/constants/theme';

export interface GlassStatCardProps {
  overline: string;
  metric: Metric<number | string>;
  delta?: { value: string; tone: 'up' | 'down' | 'flat' } | null;
  variant?: 'metric' | 'ring' | 'sparkline';
  glow?: 'you' | 'done' | 'cia';
  flush?: boolean;
  sparkline?: number[];
  testID?: string;
}

export function GlassStatCard({
  overline,
  metric,
  delta,
  variant = 'metric',
  glow,
  flush = false,
  sparkline,
  testID,
}: GlassStatCardProps): JSX.Element {
  const reduceMotion = useReducedMotion();

  // Flush mode strips all chrome (border, background, shadow) to inherit parent FrostCard.
  const baseContainerStyle = flush
    ? styles.containerFlush
    : [
        styles.container,
        glow === 'you' && styles.glowYou,
        glow === 'done' && styles.glowDone,
        glow === 'cia' && styles.glowCia,
      ];

  // KPI text styling branches by honesty state (real/low/null)
  const kpiStyle = [
    styles.kpi,
    metric.provenance.kind === 'low' && styles.kpiLow,
    metric.provenance.kind === 'null' && styles.kpiNull,
  ];

  // Delta is gated by honesty invariant: never render trend next to an unresolved (null) metric.
  const showDelta = delta !== null && delta !== undefined && metric.provenance.kind !== 'null';

  const deltaGlyph = delta?.tone === 'up' ? '▲' : delta?.tone === 'down' ? '▼' : '•';
  const deltaColor =
    delta?.tone === 'up'
      ? BalenciaColors.green
      : delta?.tone === 'down'
      ? BalenciaColors.orange
      : BalenciaColors.paper70;

  const accessibilityLabel = `${overline}: ${metric.display}${
    showDelta ? `, ${delta!.tone} ${delta!.value}` : ''
  }. ${metric.provenance.label}.`;

  return (
    <View
      testID={testID}
      accessibilityRole="summary"
      accessibilityLabel={accessibilityLabel}
      style={baseContainerStyle}
    >
      <Text style={styles.overline}>{overline}</Text>

      {variant === 'ring' ? (
        <View style={styles.chartWrapper}>
          <ProgressRing
            progress={typeof metric.value === 'number' ? metric.value : 0}
            state={metric.provenance.kind}
            glow={glow}
            centerLabel={metric.provenance.kind === 'null' ? null : metric.display}
            animateOnMount={!reduceMotion}
          />
        </View>
      ) : (
        <View style={styles.kpiRow}>
          <Text style={kpiStyle}>{metric.display}</Text>
          {showDelta && (
            <Text style={[styles.delta, { color: deltaColor }]}>
              {`${deltaGlyph} ${delta!.value}`}
            </Text>
          )}
        </View>
      )}

      {variant === 'sparkline' && (
        <View style={styles.chartWrapper}>
          <TrendChart
            points={(sparkline ?? []).map((y, i) => ({ x: i, y }))}
            state={metric.provenance.kind}
            height={56}
          />
        </View>
      )}

      {showDelta && variant === 'ring' && (
        <Text style={[styles.delta, { color: deltaColor }]}>
          {`${deltaGlyph} ${delta!.value}`}
        </Text>
      )}

      <ProvenanceChip provenance={metric.provenance} />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    gap: Spacing.two,
    padding: Spacing.three,
    borderRadius: Radius.xl,
    borderWidth: 1,
    borderColor: BalenciaColors.hairline,
    backgroundColor: BalenciaColors.inkBrown800,
  },
  containerFlush: {
    gap: Spacing.two,
    padding: Spacing.three,
    borderRadius: Radius.xl,
    backgroundColor: 'transparent',
  },
  glowYou: {
    borderColor: 'rgba(255, 94, 0, 0.32)',
    backgroundColor: BalenciaColors.orangeSoft,
    shadowColor: BalenciaColors.orange,
    shadowOffset: { width: 0, height: 0 },
    shadowOpacity: 0.2,
    shadowRadius: 8,
    elevation: 2,
  },
  glowDone: {
    borderColor: 'rgba(52, 168, 83, 0.32)',
    backgroundColor: BalenciaColors.greenSoft,
    shadowColor: BalenciaColors.green,
    shadowOffset: { width: 0, height: 0 },
    shadowOpacity: 0.2,
    shadowRadius: 8,
    elevation: 2,
  },
  glowCia: {
    borderColor: 'rgba(127, 36, 255, 0.32)',
    backgroundColor: BalenciaColors.purpleSoft,
    shadowColor: BalenciaColors.purple,
    shadowOffset: { width: 0, height: 0 },
    shadowOpacity: 0.2,
    shadowRadius: 8,
    elevation: 2,
  },
  overline: {
    fontSize: 12,
    fontWeight: '700',
    textTransform: 'uppercase',
    color: BalenciaColors.paper55,
  },
  kpiRow: {
    flexDirection: 'row',
    alignItems: 'baseline',
    gap: Spacing.two,
  },
  kpi: {
    fontSize: 32,
    fontWeight: '800',
    color: BalenciaColors.paper100,
    fontVariant: ['tabular-nums'],
  },
  kpiLow: {
    opacity: 0.64,
  },
  kpiNull: {
    color: BalenciaColors.paper55,
  },
  delta: {
    fontSize: 13,
    fontWeight: '700',
  },
  chartWrapper: {
    marginVertical: Spacing.half,
  },
});
