// GlassNavBar — floating glass-pill bottom nav (COMPACT-CANON §6,
// COMPONENT-CATALOG.md "GlassNavBar"; BIOS-004 ADR-5). Rendered as the
// `tabBar` prop of the expo-router `Tabs` navigator in `(tabs)/_layout.tsx`
// (imported from 'expo-router/js-tabs', not the deprecated native-tabs
// path — A6). Consumes `BottomTabBarProps` exactly as React Navigation's
// bottom-tabs would; that type is not installed as its own package here —
// expo-router vendors it and re-exports it from the same 'expo-router/js-tabs'
// module this component's props type comes from.
//
// Hidden-route compatibility note (ADR-5): this migration away from
// `NativeTabs` also fixes the BIOS-002 bug where `router.push()` to a
// root-level Stack screen (life-areas, fitness, data-controls, and future
// modal routes) was silently swallowed by NativeTabs' native trigger list.
// JS Tabs (this component's host navigator) does not intercept `router.push`
// to routes outside its own screen set, so those pushes now work correctly.
// This file does not add or change any root Stack route — see
// `src/app/_layout.tsx` (untouched by this packet).

type TabKey = 'today' | 'cia' | 'missions' | 'me';

type TabMeta = {
  label: string;
  sfSymbol: SFSymbol;
  sfSymbolActive: SFSymbol;
  androidSymbol: AndroidSymbol;
};

// Exact glyph/label set the old NativeTabs config used (embedded above) —
// reused verbatim so the visible IA does not change, only the chrome that
// renders it. Order matches `<Tabs.Screen>` registration in
// `(tabs)/_layout.tsx`: today, cia, missions, me.
const TAB_META: Record<TabKey, TabMeta> = {
  today: {
    label: 'Today',
    sfSymbol: 'sun.max',
    sfSymbolActive: 'sun.max.fill',
    androidSymbol: 'wb_sunny',
  },
  cia: {
    label: 'Cia',
    sfSymbol: 'sparkles',
    sfSymbolActive: 'sparkles',
    androidSymbol: 'auto_awesome',
  },
  missions: {
    label: 'Missions',
    sfSymbol: 'list.bullet.rectangle',
    sfSymbolActive: 'list.bullet.rectangle.fill',
    androidSymbol: 'checklist',
  },
  me: {
    label: 'Me',
    sfSymbol: 'person.crop.circle',
    sfSymbolActive: 'person.crop.circle.fill',
    androidSymbol: 'account_circle',
  },
};

// .glass-pill canon literals (COMPACT-CANON §2) — no matching theme.ts token
// exists yet (`chartTrack` is viz-chrome, a different value); declared local
// to this component rather than adding a token (out of scope, see packet
// "Plan-vs-live correction 3").
const GLASS_PILL_TINT = 'rgba(10, 10, 15, 0.55)'; // = ink900 @ 55% alpha
const GLASS_PILL_BORDER = 'rgba(255, 255, 255, 0.10)';

function isTabKey(name: string): name is TabKey {
  return name === 'today' || name === 'cia' || name === 'missions' || name === 'me';
}

export function GlassNavBar(props: BottomTabBarProps): JSX.Element {
  const { state, navigation, insets } = props;
  const reduceMotion = useReducedMotion();
  const glassAvailable = isGlassEffectAPIAvailable();

  const items = state.routes.map((route, index) => {
    if (!isTabKey(route.name)) return null;
    const meta = TAB_META[route.name];
    const isFocused = state.index === index;

    const onPress = () => {
      Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => undefined);
      const event = navigation.emit({
        type: 'tabPress',
        target: route.key,
        canPreventDefault: true,
      });
      if (!isFocused && !event.defaultPrevented) {
        navigation.navigate(route.name);
      }
    };

    const onLongPress = () => {
      navigation.emit({ type: 'tabLongPress', target: route.key });
    };

    return (
      <GlassNavBarItem
        key={route.key}
        routeName={route.name}
        meta={meta}
        isFocused={isFocused}
        reduceMotion={reduceMotion}
        onPress={onPress}
        onLongPress={onLongPress}
      />
    );
  });

  const pillContent = <>{items}</>;

  return (
    <View
      style={[styles.wrapper, { bottom: insets.bottom + Spacing.three }]}
      pointerEvents="box-none">
      {glassAvailable ? (
        <GlassView glassEffectStyle="regular" tintColor={GLASS_PILL_TINT} style={styles.pill}>
          {pillContent}
        </GlassView>
      ) : (
        <View style={[styles.pill, styles.pillFallback]}>{pillContent}</View>
      )}
    </View>
  );
}

function GlassNavBarItem({
  routeName,
  meta,
  isFocused,
  reduceMotion,
  onPress,
  onLongPress,
}: {
  routeName: TabKey;
  meta: TabMeta;
  isFocused: boolean;
  reduceMotion: boolean;
  onPress: () => void;
  onLongPress: () => void;
}): JSX.Element {
  const scale = useSharedValue(1);
  const animatedStyle = useAnimatedStyle(() => ({ transform: [{ scale: scale.value }] }));
  const tint = isFocused ? BalenciaColors.orange : BalenciaColors.paper55;

  // Reanimated SharedValue.value assignment from an event handler is the documented
  // mutation API (not a React state write) — the `react-hooks/immutability` ESLint
  // rule does not model SharedValue and false-positives on both writes below. This is
  // pre-existing, identical drift already present in `trend-chart.tsx`'s `onLayout`
  // handler (`measuredWidth.value = e.nativeEvent.layout.width;`, verified live) —
  // suppressed narrowly here with justification (worker-rules.md: "narrowly
  // justify+disable"), not silently left failing.
  const handlePressIn = () => {
    if (reduceMotion) return;
    // eslint-disable-next-line react-hooks/immutability
    scale.value = withTiming(0.96, {
      duration: MotionDuration.feedback,
      easing: Easing.bezier(...MotionEasing.physical),
    });
  };
  const handlePressOut = () => {
    if (reduceMotion) return;
    // eslint-disable-next-line react-hooks/immutability
    scale.value = withTiming(1, {
      duration: MotionDuration.feedback,
      easing: Easing.bezier(...MotionEasing.physical),
    });
  };

  return (
    <Pressable
      onPress={onPress}
      onLongPress={onLongPress}
      onPressIn={handlePressIn}
      onPressOut={handlePressOut}
      accessibilityRole="tab"
      accessibilityState={{ selected: isFocused }}
      accessibilityLabel={meta.label}
      testID={`glass-nav-bar-tab-${routeName}`}
      style={styles.item}>
      <Animated.View style={[styles.itemInner, animatedStyle]}>
        <SymbolView
          name={{
            ios: isFocused ? meta.sfSymbolActive : meta.sfSymbol,
            android: meta.androidSymbol,
            web: meta.androidSymbol,
          }}
          size={22}
          weight={isFocused ? 'semibold' : 'regular'}
          tintColor={tint}
        />
        <Text style={[styles.label, { color: tint, fontWeight: isFocused ? '700' : '600' }]}>
          {meta.label}
        </Text>
      </Animated.View>
    </Pressable>
  );
}

  },

import { Redirect } from 'expo-router';
import Tabs from 'expo-router/js-tabs';

import { GlassNavBar } from '@/components/balencia/glass-nav-bar';
import { useSession } from '@/services/auth/session-provider';

export default function TabsLayout() {
  const { status } = useSession();

  if (status === 'unauthenticated' || status === 'expired') {
    return <Redirect href="/(auth)/sign-in" />;
  }

  return (
    <Tabs
      screenOptions={{ headerShown: false, tabBarStyle: { display: 'none' } }}
      tabBar={(props) => <GlassNavBar {...props} />}>
      <Tabs.Screen name="today" />
      <Tabs.Screen name="cia" />
      <Tabs.Screen name="missions" />
      <Tabs.Screen name="me" />
    </Tabs>
  );
}
