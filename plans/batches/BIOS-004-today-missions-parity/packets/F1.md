// ─── Hook: useReducedMotion ─────────────────────────────────────────────
export function useReducedMotion(): boolean {
  const [reduceMotion, setReduceMotion] = useState(false);
  useEffect(() => {
    const handle = (isReducedMotion: boolean) => setReduceMotion(isReducedMotion);

### §2 — `motion.ts`: exact full file content (CREATE)

```typescript
/**
 * Shared motion constants (BIOS-004 ADR-6: "one source of truth prevents
 * drift across GLM-authored components").
 *
 * Framework-agnostic ON PURPOSE — this file exports only plain numbers,
 * tuples, and objects. It does NOT import 'react-native' or
 * 'react-native-reanimated'.
 *
 * Verified live 2026-07-09: `import { Easing } from 'react-native-reanimated'`
 * throws during Vitest collection under this project's test config
 * (vitest.config.ts -> environment: 'node'):
 *   "Cannot find module '.../react-native-worklets/lib/module/initializers/
 *    initializers' imported from .../react-native-worklets/lib/module/index.js"
 * reanimated's entrypoint reaches into the native worklets runtime, which
 * does not exist under plain Node. src/**\/*.test.ts (vitest.config.ts
 * `include`) collects ANY test that transitively imports this file, so a
 * top-level reanimated import here would break every future test that
 * touches motion.ts. Component files that actually animate (running under
 * Metro/the RN runtime, never under Vitest) build the real Reanimated easing
 * curve at the call site:
 *
 *   import { Easing } from 'react-native-reanimated';
 *   import { MotionEasing } from '@/constants/motion';
 *   const curve = Easing.bezier(...MotionEasing.physical);
 *
 * Reduced motion:
 * Gate every animated component on the existing EXPORTED hook —
 * `useReducedMotion` from '@/components/balencia/auth-inputs' (verified
 * live: exported at that path). NOTE: '@/components/balencia/states.tsx'
 * has its own private, non-exported duplicate of the same hook —
 * pre-existing drift, not introduced or fixed by this file; import the
 * auth-inputs one.
 * When `useReducedMotion()` is true: skip straight to the FINAL static state
 * — no sweep, draw, stagger, spring, or breathe — but text value updates and
 * haptics still fire (canon: "text updates and haptics still occur").
 * Breathing collapses to a static glow (no pulse). Animate `transform` and
 * `opacity` ONLY, never color/blur/shadow (ADR-6, binding).
 */

/** Duration constants, milliseconds. Every value is traceable to a specific
 *  hi-fi spec line or ADR-6 — see the per-key comment. */
export const MotionDuration = {
  /** General UI feedback range (ADR-6: "feedback 150-250ms"). Use `feedback`
   *  as the default unless a more specific value below applies. */
  feedbackMin: 150,
  feedbackMax: 250,
  feedback: 200,
  /** Hero mount sweep — ProgressRing 0->value on mount (ADR-6: "sweep
   *  520ms"; spec S14: "Hero ring: 0->72% sweep on mount (520ms physical
   *  ease-out)"). */
  sweep: 520,
  /** Re-sweep on completion — shorter than the mount sweep (spec S14: "on
   *  completion, shorter 250ms re-sweep"). */
  sweepShort: 250,
  /** Stroke/line draw, left-to-right (ADR-6: "draw 400ms"; spec S42:
   *  "stroke draws L->R ~400ms"). TrendChart line draw, celebration divider. */
  draw: 400,
  /** ExpandableList accordion open/close (spec S14: "Accordion: spring
   *  200ms, chevron rotates 180 degrees"). Duration-only in spec — see
   *  MotionSpring's doc comment for why this is NOT a withSpring config. */
  accordion: 200,
  /** IntelligenceTimeline per-stage reveal inside the CIA Reasoning
   *  accordion (spec S14: "stages at ~180ms intervals"). */
  timelineStage: 180,
  /** Stalled-mission Sheet (spec S14: "Sheet (stalled): spring 250ms"). */
  sheetSpring: 250,
  /** Missions board Summary Band fill, left-to-right (spec S13: "Board
   *  Summary Band fills L->R (250ms)"). Also used for the S13 Radar draw-in
   *  ("Radar draws in (250ms)"). */
  bandFill: 250,
  /** Shared stagger step for cascading list/section entries. Not enumerated
   *  in ADR-6 itself — derived from two independent composition specs that
   *  agree on the same value: spec S13 ("Mission cards cascade downward
   *  (80ms stagger)") and the architecture plan's Today composition map
   *  (§4.1: "Motion: staggered entries 80ms"). */
  stagger: 80,
  /** Next Action card swap on completion — old card out, replacement in
   *  (spec S14: "250ms ease-in-out"). */
  actionSwap: 250,
  /** Celebration toast slide-down-in (spec S42 Motion: "slide-down-in
   *  200ms"). */
  toastIn: 200,
  /** Celebration toast slide-up-out on dismiss (spec S42 States /
   *  fast-dismiss: "slide-up-out 150ms"). */
  toastOut: 150,
  /** S15 create-mission input -> processing shrink (spec S15 Motion:
   *  "Input->processing shrink (280ms)"). */
  processingShrink: 280,
  /** S15 save success glow-done swap (spec S15 Motion: "Save success:
   *  200ms glow-done swap + success haptic"). */
  successSwap: 200,
} as const;

/**
 * Easing curves as plain bezier control-point tuples — NOT Reanimated
 * `EasingFn` values (see file header for why). Tuple order matches
 * `Easing.bezier(x1, y1, x2, y2)`; spread at the call site.
 */
export const MotionEasing = {
  /** Canon physical easing (COMPACT-CANON §6: "physical easing (never
   *  linear)"; ADR-6: "cubic-bezier(0.32,0.72,0,1)"). Default for
   *  sweeps/draws/entrances unless a spec calls for a different curve. */
  physical: [0.32, 0.72, 0, 1],
} as const;

/**
 * Reanimated `withSpring` config objects — plain data, no reanimated import
 * required to define them. ONLY `badge` has a spec-given physics config:
 * spec S42 "Entrance: spring physics (stiffness 150, damping 15) for badge
 * scale-in" (ADR-6 restates the same pair). Every OTHER "spring Nms" mention
 * in the hi-fi specs (accordion, stalled sheet) gives a settle DURATION, not
 * stiffness/damping — inventing physics values for those would fabricate
 * precision the source docs do not give. Use
 * `withTiming(value, { duration: MotionDuration.accordion, easing: Easing.bezier(...MotionEasing.physical) })`
 * for accordion/sheet, not `withSpring`.
 */
export const MotionSpring = {
  badge: { stiffness: 150, damping: 15 },
} as const;

/**
 * Hero-card glow breathe cycle (ADR-6: "breathe (4s, 55%<->100%)"). Apply to
 * exactly ONE hero card per screen — spec S14 Motion: "Glow breathe: shared
 * 4s cycle across all glows (one rhythm, not five uncoordinated pulses)".
 * NOTE: spec S42's celebration FrostCard breathe is a documented per-screen
 * exception ("glow-done breathes (60%->100% opacity, 3s ease)") — that value
 * belongs to the celebration composition packet (wave 4c), not this shared
 * default; do not overwrite these numbers to match it.
 */
export const MotionBreathe = {
  durationMs: 4000,
  opacityMin: 0.55,
  opacityMax: 1,
} as const;
```
