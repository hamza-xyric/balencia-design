/**
 * Pure logic for the missions board (S13 parity, packet S2). No
 * react-native / reanimated / '@/constants/theme' imports — this file must
 * stay safely importable under Vitest's node environment (see
 * constants/motion.ts header: any transitive react-native-reanimated import
 * breaks src/**\/*.test.ts collection). Only TYPES are imported from
 * services/adapters/missions (a type-only import erases at compile time, so
 * it pulls in no runtime react-native dependency).
 */
import type { MissionTier, MissionVM } from '@/services/adapters/missions';

export type StatusTab = 'active' | 'done' | 'all';

/** Canon tier order — matches `MissionMetalColors` key order in
 *  `constants/theme.ts` (life/main/side/weekly/daily/group). The screen
 *  renders one ChipDomainTag per entry, including 'group' — ADR-4: no
 *  mission ever maps to 'group' this batch (no server squad/group signal),
 *  so the chip renders honestly inert rather than being hidden or faked. */
export const ALL_TIERS: readonly MissionTier[] = [
  'life',
  'main',
  'side',
  'weekly',
  'daily',
  'group',
] as const;

const TIER_LABELS: Record<MissionTier, string> = {
  life: 'Life mission',
  main: 'Main mission',
  side: 'Side mission',
  weekly: 'Weekly mission',
  daily: 'Daily mission',
  group: 'Group mission',
};

/** Sentence-case RPG tier label (canon copy rule: sentence case, no
 *  exclamation marks). Pure lookup — never guesses a tier outside the map. */
export function tierLabel(tier: MissionTier): string {
  return TIER_LABELS[tier];
}

/**
 * Status-tab + tier-chip filter, applied together (AND).
 *
 * Status:
 * - 'active' matches VM status 'active' OR 'blocked' (missions.ts mapStatus:
 *   'blocked' means paused-but-still-on-board, not done).
 * - 'done' matches VM status 'complete' only.
 * - 'all' passes every status.
 *
 * Tier: an EMPTY tierFilter means NO tier filtering (every tier passes) —
 * that is the default/unfiltered state, never "match nothing".
 */
export function filterMissions(
  missions: readonly MissionVM[],
  statusTab: StatusTab,
  tierFilter: readonly MissionTier[],
): MissionVM[] {
  return missions.filter((mission) => {
    const statusMatch =
      statusTab === 'all' ||
      (statusTab === 'done' && mission.status === 'complete') ||
      (statusTab === 'active' && (mission.status === 'active' || mission.status === 'blocked'));
    if (!statusMatch) {
      return false;
    }
    return tierFilter.length === 0 || tierFilter.includes(mission.tier);
  });
}

export interface LevelProgressInput {
  currentXP: number;
  xpForNextLevel: number;
}

/**
 * Honest "XP to next level" copy for the MomentumBar label.
 *
 * `null` input (the gamification-stats query has not resolved) returns
 * `null` — the caller renders the MomentumBar honest-null state. NEVER
 * fabricates a number when the source query has not confirmed one.
 */
export function xpToNextLabel(levelProgress: LevelProgressInput | null): string | null {
  if (levelProgress === null) {
    return null;
  }
  const remaining = Math.max(0, levelProgress.xpForNextLevel - levelProgress.currentXP);
  return `${remaining} XP to next level`;
}

import { describe, it, expect } from 'vitest';

import type { MissionTier, MissionVM } from '@/services/adapters/missions';
import { ALL_TIERS, filterMissions, tierLabel, xpToNextLabel } from './missions-board-logic';

function makeMission(overrides: Partial<MissionVM> = {}): MissionVM {
  return {
    id: 'm1',
    title: 'Test mission',
    description: null,
    tier: 'side',
    status: 'active',
    progress: 40,
    domain: 'fitness',
    source: 'life',
    isPrimary: false,
    dueDate: null,
    target: null,
    ...overrides,
  };
}

describe('filterMissions', () => {
  const missions: MissionVM[] = [
    makeMission({ id: 'a', tier: 'life', status: 'active' }),
    makeMission({ id: 'b', tier: 'main', status: 'blocked' }),
    makeMission({ id: 'c', tier: 'side', status: 'complete' }),
    makeMission({ id: 'd', tier: 'weekly', status: 'active' }),
  ];

  it('active tab matches active and blocked statuses, excludes complete', () => {
    const result = filterMissions(missions, 'active', []);
    expect(result.map((m) => m.id).sort()).toEqual(['a', 'b', 'd']);
  });

  it('done tab matches only complete', () => {
    const result = filterMissions(missions, 'done', []);
    expect(result.map((m) => m.id)).toEqual(['c']);
  });

  it('all tab passes every status', () => {
    const result = filterMissions(missions, 'all', []);
    expect(result).toHaveLength(4);
  });

  it('an empty tier filter applies no tier filtering', () => {
    const activeResult = filterMissions(missions, 'active', []);
    const allResult = filterMissions(missions, 'all', []);
    expect(activeResult).toHaveLength(3);
    expect(allResult).toHaveLength(4);
  });

  it('a multi-select tier filter narrows to the selected tiers only', () => {
    const result = filterMissions(missions, 'all', ['life', 'weekly']);
    expect(result.map((m) => m.id).sort()).toEqual(['a', 'd']);
  });

  it('combines status and tier filters with AND semantics', () => {
    const result = filterMissions(missions, 'active', ['main']);
    expect(result.map((m) => m.id)).toEqual(['b']);
  });

  it('a tier with zero matching missions (group) returns empty, honestly', () => {
    const result = filterMissions(missions, 'all', ['group']);
    expect(result).toEqual([]);
  });

  it('is pure — does not mutate the input array', () => {
    const before = [...missions];
    filterMissions(missions, 'active', ['life']);
    expect(missions).toEqual(before);
  });
});

describe('tierLabel', () => {
  it('has exactly six canon tiers, in life/main/side/weekly/daily/group order', () => {
    expect(ALL_TIERS).toEqual(['life', 'main', 'side', 'weekly', 'daily', 'group']);
  });

  it('maps every tier to its sentence-case RPG label', () => {
    const expected: Record<MissionTier, string> = {
      life: 'Life mission',
      main: 'Main mission',
      side: 'Side mission',
      weekly: 'Weekly mission',
      daily: 'Daily mission',
      group: 'Group mission',
    };
    ALL_TIERS.forEach((tier) => {
      expect(tierLabel(tier)).toBe(expected[tier]);
    });
  });
});

describe('xpToNextLabel', () => {
  it('returns null (honest-null) when levelProgress is null', () => {
    expect(xpToNextLabel(null)).toBeNull();
  });

  it('formats remaining XP for a real levelProgress', () => {
    expect(xpToNextLabel({ currentXP: 380, xpForNextLevel: 800 })).toBe('420 XP to next level');
  });

  it('never returns a negative remaining value', () => {
    expect(xpToNextLabel({ currentXP: 900, xpForNextLevel: 800 })).toBe('0 XP to next level');
  });

  it('handles a zero-progress level start', () => {
    expect(xpToNextLabel({ currentXP: 0, xpForNextLevel: 500 })).toBe('500 XP to next level');
  });
});

import { useEffect, useMemo, useState } from 'react';

  useAnimatedStyle,
  useSharedValue,
  withDelay,
  withTiming,
} from 'react-native-reanimated';
import { useRouter, type Href } from 'expo-router';

import { MotionDuration, MotionEasing } from '@/constants/motion';
import { nullMetric, realMetric } from '@/services/adapters/provenance';
import type { MissionTier, MissionVM } from '@/services/adapters/missions';
import { useMissions } from '@/hooks/use-missions';
import { useGamification } from '@/hooks/use-gamification';

import { ALL_TIERS, filterMissions, tierLabel, xpToNextLabel, type StatusTab } from './missions-board-logic';

const statusTabs: { key: StatusTab; label: string }[] = [
  { key: 'active', label: 'active' },
  { key: 'done', label: 'done' },
  { key: 'all', label: 'all' },
];

const statusChipTone: Record<MissionVM['status'], 'orange' | 'warning' | 'green'> = {
  active: 'orange',
  blocked: 'warning',
  complete: 'green',
};

export function MissionsScreen() {
  const router = useRouter();
  const reduceMotion = useReducedMotion();
  const { data, status, refetch } = useMissions();
  const { data: gamification, status: gamificationStatus } = useGamification();

  const [statusTab, setStatusTab] = useState<StatusTab>('active');
  const [tierFilter, setTierFilter] = useState<MissionTier[]>([]);
  const [filtersOpen, setFiltersOpen] = useState(false);

  const toggleTier = (tier: MissionTier) => {
    setTierFilter((current) =>
      current.includes(tier) ? current.filter((t) => t !== tier) : [...current, tier],
    );
  };

  const filteredMissions = useMemo(
    () => (data ? filterMissions(data.missions, statusTab, tierFilter) : []),
    [data, statusTab, tierFilter],
  );

  const fab = (
    <FABQuickLog
      onPress={() => router.push('/mission-create' as Href)}
      accessibilityLabel="Create mission"
      testID="missions-fab"
    />
  );

  if (status === 'pending') {
    return (
      <View style={styles.root}>
        <BalenciaScreen eyebrow="Mission board" title="your missions">
          <SkeletonState variant="list" lines={3} />
        </BalenciaScreen>
        {fab}
      </View>
    );
  }

  if (status === 'error' || !data) {
    return (
      <View style={styles.root}>
        <BalenciaScreen eyebrow="Mission board" title="your missions">
          <ErrorState message="Couldn't load your missions." onRetry={refetch} />
        </BalenciaScreen>
        {fab}
      </View>
    );
  }

  const activeMetric = realMetric(data.summary.active, {
    source: 'server',
    label: 'via missions sync',
  });
  const doneMetric = realMetric(data.summary.completed, {
    source: 'server',
    label: 'via missions sync',
  });
  const gamificationReady = gamificationStatus === 'success' && gamification !== undefined;
  const streakMetric = gamificationReady
    ? realMetric(gamification.streak, { source: 'server', label: 'via gamification stats' })
    : nullMetric('Not enough data yet');
  const levelProgress = gamificationReady ? gamification.levelProgress : null;
  const xpLabel = xpToNextLabel(levelProgress);
  const xpValue = levelProgress?.currentXP ?? 0;
  const xpMax = levelProgress?.xpForNextLevel ?? 100;

  const isBoardEmpty = data.missions.length === 0;

  return (
    <View style={styles.root}>
      <BalenciaScreen
        eyebrow="Mission board"
        title="your missions"
        subtitle="Missions are mapped from unified goals when signed in.">
        <Pressable
          onPress={() => setFiltersOpen((open) => !open)}
          accessibilityRole="button"
          accessibilityLabel={filtersOpen ? 'Hide mission filters' : 'Show mission filters'}
          hitSlop={8}
          style={({ pressed }) => [styles.filterToggle, pressed && styles.filterTogglePressed]}
          testID="missions-filter-toggle">
          <Text
            style={styles.filterToggleGlyph}
            accessibilityElementsHidden
            importantForAccessibility="no">
            ▤
          </Text>
          <Text style={styles.filterToggleLabel}>{filtersOpen ? 'Hide filters' : 'Filter'}</Text>
        </Pressable>

        <View style={styles.summaryBand}>
          <View style={styles.summaryTile}>
            <GlassStatCard overline="Active" metric={activeMetric} glow="you" testID="missions-summary-active" />
          </View>
          <View style={styles.summaryTile}>
            <GlassStatCard overline="Done" metric={doneMetric} glow="done" testID="missions-summary-done" />
          </View>
          <View style={styles.summaryTile}>
            <GlassStatCard overline="Streak" metric={streakMetric} glow="you" testID="missions-summary-streak" />
          </View>
        </View>

        <MomentumBar
          value={xpValue}
          max={xpMax}
          state={xpLabel === null ? 'null' : 'real'}
          label={xpLabel ?? 'Not enough data yet'}
          glow="you"
          testID="missions-xp-momentum"
        />

        <SegmentedTabs
          options={statusTabs}
          value={statusTab}
          onChange={setStatusTab}
          accessibilityLabel="Filter missions by status"
          testID="missions-status-tabs"
        />

        {filtersOpen && (
          <TierFilterRow tierFilter={tierFilter} onToggleTier={toggleTier} reduceMotion={reduceMotion} />
        )}

        <View
          accessible
          accessibilityRole="text"
          accessibilityLabel="Life areas radar, arrives with life areas">
          <GlassCard>
            <CardTitle>Life areas</CardTitle>
            <View style={styles.radarGhostRing} />
            <CardBody>Constellation radar arrives with life areas</CardBody>
          </GlassCard>
        </View>

        {filteredMissions.length === 0 ? (
          <HonestNullState
            title={isBoardEmpty ? 'No missions yet' : 'No missions here yet'}
            detail={
              isBoardEmpty
                ? 'Create your first mission with Cia.'
                : 'Try a different status or tier filter.'
            }
          />
        ) : (
          <View style={styles.list}>
            {filteredMissions.map((mission, index) => (
              <MissionRow
                key={mission.id}
                mission={mission}
                index={index}
                reduceMotion={reduceMotion}
                onPress={() => router.push(`/mission/${mission.id}` as Href)}
                testID={`missions-row-${mission.id}`}
              />
            ))}
          </View>
        )}
      </BalenciaScreen>
      {fab}
    </View>
  );
}

function TierFilterRow({
  tierFilter,
  onToggleTier,
  reduceMotion,
}: {
  tierFilter: MissionTier[];
  onToggleTier: (tier: MissionTier) => void;
  reduceMotion: boolean;
}) {
  const progress = useSharedValue(reduceMotion ? 1 : 0);

  useEffect(() => {
    if (reduceMotion) {
      progress.value = 1;
    } else {
      progress.value = withTiming(1, {
        duration: MotionDuration.accordion,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }
  }, [reduceMotion, progress]);

  const animatedStyle = useAnimatedStyle(() => ({
    opacity: progress.value,
    transform: [{ translateY: (1 - progress.value) * -6 }],
  }));

  return (
    <Animated.View style={[styles.tierRow, animatedStyle]} testID="missions-tier-filter-row">
      {ALL_TIERS.map((tier) => (
        <ChipDomainTag
          key={tier}
          label={tierLabel(tier)}
          active={tierFilter.includes(tier)}
          onPress={() => onToggleTier(tier)}
          testID={`missions-tier-chip-${tier}`}
        />
      ))}
    </Animated.View>
  );
}

function MissionRow({
  mission,
  index,
  reduceMotion,
  onPress,
  testID,
}: {
  mission: MissionVM;
  index: number;
  reduceMotion: boolean;
  onPress: () => void;
  testID?: string;
}) {
  const progress = useSharedValue(reduceMotion ? 1 : 0);

  useEffect(() => {
    if (reduceMotion) {
      progress.value = 1;
    } else {
      progress.value = withDelay(
        index * MotionDuration.stagger,
        withTiming(1, {
          duration: MotionDuration.feedback,
          easing: Easing.bezier(...MotionEasing.physical),
        }),
      );
    }
  }, [reduceMotion, index, progress]);

  const animatedStyle = useAnimatedStyle(() => ({
    opacity: progress.value,
    transform: [{ translateY: (1 - progress.value) * -8 }],
  }));

  const metal = MissionMetalColors[mission.tier];
  const accessibilityLabel = `${mission.title}, ${tierLabel(mission.tier)}, ${Math.round(mission.progress)} percent complete`;

  return (
    <Animated.View style={[styles.rowWrap, { borderLeftColor: metal.solid }, animatedStyle]}>
      <Pressable
        onPress={onPress}
        accessibilityRole="button"
        accessibilityLabel={accessibilityLabel}
        style={({ pressed }) => [pressed && styles.rowPressed]}
        testID={testID}>
        <GlassCard tone={mission.status === 'complete' ? 'green' : 'default'}>
          <View style={styles.rowContent}>
            <ProgressRing
              progress={mission.progress}
              size={36}
              stroke={4}
              glow={mission.status === 'complete' ? 'done' : 'you'}
              testID={testID ? `${testID}-ring` : undefined}
            />
            <View style={styles.rowBody}>
              <Text style={styles.rowTitle}>{mission.title}</Text>
              <Text style={styles.rowDomain}>{mission.domain}</Text>
              <View style={styles.rowChips}>
                <View style={[styles.tierBadge, { backgroundColor: metal.soft }]}>
                  <Text style={[styles.tierBadgeText, { color: metal.solid }]}>{tierLabel(mission.tier)}</Text>
                </View>
                <StatusChip label={mission.status} tone={statusChipTone[mission.status]} />
              </View>
            </View>
          </View>
        </GlassCard>
      </Pressable>
    </Animated.View>
  );
}
