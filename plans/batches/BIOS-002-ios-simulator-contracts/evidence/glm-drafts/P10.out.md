import { StyleSheet, Text, View } from 'react-native';
import { useRouter } from 'expo-router';

import {
  BalenciaScreen,
  CardBody,
  CardTitle,
  GlassCard,
  MetricTile,
  ProvenanceChip,
  SkeletonState,
  ErrorState,
  HonestNullState,
  StatusChip,
} from '@/components/balencia';
import { BalenciaColors, Spacing } from '@/constants/theme';
import { useOverview } from '@/hooks/use-overview';

export function TodayScreen() {
  const router = useRouter();
  const { data, status, refetch } = useOverview();

  if (status === 'pending') {
    return (
      <BalenciaScreen eyebrow="Today" title="Today">
        <SkeletonState variant="hero" />
      </BalenciaScreen>
    );
  }

  if (status === 'error' || !data) {
    return (
      <BalenciaScreen eyebrow="Today" title="Today">
        <ErrorState message="Couldn't load today's briefing." onRetry={refetch} />
      </BalenciaScreen>
    );
  }

  const lifePower = data.lifePower;
  const pulse = data.pulse ?? [];
  const priorities = data.priorities ?? [];
  const momentum = data.momentum ?? [];
  const hasBrief = Boolean(data.headline);

  return (
    <BalenciaScreen
      eyebrow="Live backend"
      title="Today"
      subtitle="A native control room for Life Power, Missions, Cia, and trust status.">
      <GlassCard tone="orange">
        {lifePower.value === null ? (
          <HonestNullState
            title="Life Power isn't calibrated yet"
            detail="Track a few days and Cia will map it."
          />
        ) : (
          <View style={styles.heroRow}>
            <View style={styles.lifePower}>
              <Text style={styles.lifePowerValue}>{lifePower.display}</Text>
              <Text style={styles.lifePowerLabel}>Life Power</Text>
            </View>
            <View style={styles.heroCopy}>
              <CardTitle>Daily Momentum</CardTitle>
              <CardBody>
                Cia recommends protecting the next energy block.
              </CardBody>
            </View>
          </View>
        )}
        <ProvenanceChip provenance={lifePower.provenance} />
      </GlassCard>

      <GlassCard>
        <CardTitle>Daily Pulse</CardTitle>
        {pulse.length === 0 ? (
          <HonestNullState title="No pulse metrics available" />
        ) : (
          <View style={styles.metrics}>
            {pulse.map((metric) => (
              <MetricTile
                key={metric.label}
                label={metric.label}
                value={metric.display}
              />
            ))}
          </View>
        )}
        {pulse.map((metric) => (
          <View key={`${metric.label}-prov`} style={styles.provenanceRow}>
            <Text style={styles.metricLabel}>{metric.label}</Text>
            <ProvenanceChip provenance={metric.provenance} />
          </View>
        ))}
      </GlassCard>

      {hasBrief && (
        <GlassCard tone="purple">
          <CardTitle>Cia briefing</CardTitle>
          <CardBody>{data.headline}</CardBody>
          <StatusChip label="Cia" tone="purple" />
        </GlassCard>
      )}

      <GlassCard>
        <CardTitle>Top Priorities</CardTitle>
        {priorities.length === 0 ? (
          <HonestNullState title="No priorities yet" detail="Cia will map them soon." />
        ) : (
          priorities.map((priority) => (
            <CardBody key={priority.id}>{priority.title}</CardBody>
          ))
        )}
      </GlassCard>

      <GlassCard>
        <CardTitle>Momentum</CardTitle>
        <View style={styles.metrics}>
          {momentum.map((metric) => (
            <MetricTile
              key={metric.label}
              label={metric.label}
              value={metric.display}
            />
          ))}
        </View>
      </GlassCard>

      <GlassCard>
        <CardTitle>Trust gates</CardTitle>
        <CardBody>Ready when you are.</CardBody>
        <StatusChip label="Active" tone="green" />
        <BalenciaButton tone="quiet" onPress={() => router.push('/(tabs)/data-controls')}>
          Data controls
        </BalenciaButton>
      </GlassCard>
    </BalenciaScreen>
  );
}

import { StyleSheet, Text, View } from 'react-native';

import {
  BalenciaScreen,
  CardBody,
  CardTitle,
  GlassCard,
  StatBar,
  StatusChip,
  SkeletonState,
  ErrorState,
  HonestNullState,
} from '@/components/balencia';
import { BalenciaColors, Spacing } from '@/constants/theme';
import { useMissions } from '@/hooks/use-missions';

const tierTone = {
  life: 'orange',
  main: 'muted',
  side: 'warning',
  daily: 'green',
  weekly: 'purple',
} as const;

const statusTone = {
  active: 'green',
  blocked: 'warning',
  complete: 'muted',
} as const;

export function MissionsScreen() {
  const { data, status, refetch } = useMissions();

  if (status === 'pending') {
    return (
      <BalenciaScreen eyebrow="Mission Board" title="Missions">
        <SkeletonState variant="list" lines={3} />
      </BalenciaScreen>
    );
  }

  if (status === 'error' || !data) {
    return (
      <BalenciaScreen eyebrow="Mission Board" title="Missions">
        <ErrorState message="Couldn't load missions." onRetry={refetch} />
      </BalenciaScreen>
    );
  }

  if (data.missions.length === 0) {
    return (
      <BalenciaScreen eyebrow="Mission Board" title="Missions">
        <HonestNullState title="No missions yet" detail="Create your first mission with Cia." />
      </BalenciaScreen>
    );
  }

  return (
    <BalenciaScreen
      eyebrow="Mission Board"
      title="Missions"
      subtitle="Missions are mapped from unified goals when signed in.">
      <GlassCard>
        <View style={styles.row}>
          <CardTitle>Summary</CardTitle>
          <StatusChip label={`${data.summary.active} active`} tone="orange" />
          <StatusChip label={`${data.summary.complete} complete`} tone="green" />
          <StatusChip label={`${data.summary.blocked} blocked`} tone="warning" />
        </View>
      </GlassCard>

      {data.missions.map((mission) => (
        <GlassCard key={mission.id} tone={mission.status === 'complete' ? 'green' : 'default'}>
          <View style={styles.row}>
            <CardTitle>{mission.title}</CardTitle>
            <StatusChip label={mission.tier} tone={tierTone[mission.tier]} />
          </View>
          <CardBody>{mission.domain} domain, {mission.xp} XP reward</CardBody>
          <StatBar 
            label={mission.status} 
            value={mission.progress} 
            tone={mission.status === 'complete' ? 'green' : 'orange'} 
          />
          <Text style={styles.status}>
            {mission.status === 'blocked' 
              ? 'Blocked by a readiness or data dependency' 
              : 'Trackable in the pilot lane'}
          </Text>
          <StatusChip label={mission.status} tone={statusTone[mission.status]} />
        </GlassCard>
      ))}
    </BalenciaScreen>
  );
}

const styles = StyleSheet.create({
  row: {
    flexDirection: 'row',
    justifyContent: 'space-between',
    alignItems: 'flex-start',
    gap: Spacing.two,
    flexWrap: 'wrap',
  },
  status: {
    color: BalenciaColors.paper55,
    fontSize: 12,
    fontWeight: '700',
  },
});
