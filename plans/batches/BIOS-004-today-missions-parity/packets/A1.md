```ts
import type { UnifiedGoalsResponseDto, UnifiedGoalDto, GoalSource } from '@/services/api/dto/goals';

export type MissionTier = 'life' | 'main' | 'side';

export type MissionVM = {
  id: string;
  title: string;
  description: string | null;
  tier: MissionTier;
  status: 'active' | 'blocked' | 'complete';
  progress: number;
  domain: string;
  source: GoalSource;
  isPrimary: boolean;
  dueDate: string | null;
  target: { current: number; target: number; unit: string } | null;
};

export type MissionsVM = {
  missions: MissionVM[];
  summary: {
    total: number;
    active: number;
    completed: number;
    avgProgress: number;
  };
};

function mapTier(goal: UnifiedGoalDto): MissionTier {
  if (goal.isPrimary && goal.source === 'life') {
    return 'life';
  }
  if (goal.isPrimary) {
    return 'main';
  }
  return 'side';
}

function mapStatus(goal: UnifiedGoalDto): MissionVM['status'] {
  switch (goal.status) {
    case 'active':
    case 'in_progress':
      return 'active';
    case 'paused':
      return 'blocked';
    case 'completed':
      return 'complete';
    default:
      return 'active';
  }
}

export function toMissionsView(dto: UnifiedGoalsResponseDto): MissionsVM {
  const goals = dto.goals ?? [];
  
  const missions: MissionVM[] = goals.map((goal) => {
    return {
      id: goal.id,
      title: goal.title,
      description: goal.description ?? null,
      tier: mapTier(goal),
      status: mapStatus(goal),
      progress: goal.progress ?? 0,
      domain: goal.domain ?? 'general',
      source: goal.source,
      isPrimary: goal.isPrimary ?? false,
      dueDate: goal.dueDate ?? null,
      target: goal.target ?? null,
    };
  });

  const summary = dto.summary ?? {
    total: 0,
    active: 0,
    inProgress: 0,
    paused: 0,
    completed: 0,
    avgProgress: 0,
    bySource: {},
  };

  return {
    missions,
    summary: {
      total: summary.total ?? 0,
      active: (summary.active ?? 0) + (summary.inProgress ?? 0),
      completed: summary.completed ?? 0,
      avgProgress: summary.avgProgress ?? 0,
    },
  };
}
```

```ts
import { useQuery } from '@tanstack/react-query';

import { apiFetch } from '@/services/api/client';
import type { UnifiedGoalsResponseDto } from '@/services/api/dto/goals';
import { queryKeys } from '@/services/query/keys';
import { useSession } from '@/services/auth/session-provider';
import { toMissionsView } from '@/services/adapters/missions';

export function useMissions() {
  const { status } = useSession();

  return useQuery({
    queryKey: queryKeys.missions(),
    queryFn: () => apiFetch<UnifiedGoalsResponseDto>('/v1/goals/unified'),
    enabled: status === 'authenticated' || status === 'refreshing',
    staleTime: 60_000, // 1 minute
    select: toMissionsView,
  });
}
```

The raw `UnifiedGoalsResponseDto` (goal array + summary) is what lives in the
`queryKeys.missions()` cache entry — `toMissionsView` (`select`) runs per-observer,
**on top of** the cached raw data, so `queryClient.getQueryData(queryKeys.missions())`
in this packet's mutations hook returns the raw, un-selected `UnifiedGoalsResponseDto`,
not a `MissionsVM`. This is why the optimistic patch below operates on
`UnifiedGoalDto[]`, not `MissionVM[]`.

```ts
const mutation = useMutation<
  PrivacyPatchResponseDto,
  ApiError,
  Partial<PrivacyPreferencesDto>,
  { previousData?: PreferencesResponseDto }
>({
  mutationFn: (patch: Partial<PrivacyPreferencesDto>) => updatePrivacy(patch),
  onMutate: async (patch: Partial<PrivacyPreferencesDto>) => {
    await queryClient.cancelQueries({ queryKey: queryKeys.preferences() });
    const previousData = queryClient.getQueryData<PreferencesResponseDto>(queryKeys.preferences());
    if (previousData) {
      const optimisticData: PreferencesResponseDto = { ...previousData, preferences: { ...previousData.preferences, privacy: { ...previousData.preferences.privacy, ...patch } } };
      queryClient.setQueryData<PreferencesResponseDto>(queryKeys.preferences(), optimisticData);
    }
    return { previousData };
  },
  onError: (_error, _patch, context) => {
    if (context?.previousData) {
      queryClient.setQueryData<PreferencesResponseDto>(queryKeys.preferences(), context.previousData);
    }
  },
  onSettled: () => {
    queryClient.invalidateQueries({ queryKey: queryKeys.preferences() });
  },
});
```

Replace the `MissionTier` type and `mapTier` function in

```ts
export type MissionTier = 'life' | 'main' | 'side' | 'weekly' | 'daily' | 'group';

// ADR-4 6-tier derivation — priority-ordered, first match wins. Every branch is
// a REAL server signal (isPrimary, source, cadence) from UnifiedGoalDto; nothing
// here is guessed. 'group' is never returned this batch (ADR-4: no server
// squad/group signal exists to derive it from) — the branch simply does not
// exist below; MissionMetalColors.group in constants/theme.ts stays a forward-
// compat token with nothing rendering it yet.
export function mapTier(goal: UnifiedGoalDto): MissionTier {
  if (goal.isPrimary && goal.source === 'life') {
    return 'life';
  }
  if (goal.isPrimary) {
    return 'main';
  }
  // NOTE (honesty caveat, keep as a code comment): `goal.cadence` is NOT a
  // schedule field server-side — server/src/services/goals-aggregator/
  // goals-aggregator.service.ts:41,61 populate it from `target_unit` (health)
  // /`target_unit` (life) — a measurement unit like "kg" or "reps", not a
  // cadence enum. career/finance goals always send cadence:null (:79,99). This
  // branch is a REAL, honest signal (never fabricated) but will rarely fire in
  // practice — it only matches if a goal's target_unit literally equals the
  // string "weekly"/"daily". This is the exact, documented signal ADR-4 chose;
  // do not "improve" it by inferring cadence from anything else.
  if (goal.cadence === 'weekly') {
    return 'weekly';
  }
  if (goal.cadence === 'daily') {
    return 'daily';
  }
  return 'side';
}
```

### 5. `src/hooks/use-mission-mutations.ts` (CREATE — full file)

```ts
import { useMutation, useQueryClient, type QueryClient } from '@tanstack/react-query';

import { apiFetch } from '@/services/api/client';
import { ApiError } from '@/services/api/errors';
import { createLifeMission, completeLifeMission, completeHealthMission } from '@/services/api/missions';
import type {
  CreateLifeGoalRequestDto,
  LifeGoalResponseDto,
  LifeGoalDto,
  HealthGoalDto,
  UnifiedGoalsResponseDto,
  UnifiedGoalDto,
} from '@/services/api/dto/goals';
import type { GamificationStatsResponseDto } from '@/services/api/dto/gamification';
import { queryKeys } from '@/services/query/keys';

export type CreateMissionInput = CreateLifeGoalRequestDto;

export type CompleteMissionInput = {
  id: string;
  /** Only life+health have a wired complete surface this batch (ADR-4). */
  source: 'life' | 'health';
};

export type CompleteMissionResult = {
  goal: LifeGoalDto | HealthGoalDto;
  /**
   * Confirmed POSITIVE XP delta only, or `null`. `null` covers: no
   * pre-complete gamification snapshot was cached yet, the refetched total
   * is unchanged (the common case — life/health XP is usually an async
   * ≤15-min achievement job, ADR-4), or (defensively) a non-positive delta.
   * Never fabricated — a caller (S42 celebration, future packet) must only
   * render a reward state when this is a positive number.
   */
  xpDelta: number | null;
};

type MissionsSnapshot = { previousData?: UnifiedGoalsResponseDto };

async function fetchGamificationStats(queryClient: QueryClient) {
  return queryClient.fetchQuery<GamificationStatsResponseDto>({
    queryKey: queryKeys.gamification(),
    queryFn: () => apiFetch<GamificationStatsResponseDto>('/gamification/stats'),
    // Force a real network round-trip — the 60s staleTime configured in
    // services/query/client.ts / use-gamification.ts must never mask a
    // just-happened XP change behind a stale cached read.
    staleTime: 0,
  });
}

/**
 * ADR-4 primary in-scope CREATE surface (life missions only — health/career/
 * finance create are documented-deferred). No optimistic insert: the server
 * assigns the id and can silently SMART-rewrite title/description before
 * persisting (life-goals.service.ts:130-149) — inserting an optimistic row
 * risks showing text the server didn't actually save. `onSettled` refetches
 * the real list instead.
 */
export function useCreateMission() {
  const queryClient = useQueryClient();

  return useMutation<LifeGoalResponseDto, ApiError, CreateMissionInput>({
    mutationFn: (input) => createLifeMission(input),
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: queryKeys.missions() });
    },
  });
}

/**
 * ADR-4 complete surface (life + health) with:
 * - optimistic local status→'completed'/progress→100 flip + snapshot/rollback
 *   (ADR-4 "Decision — optimistic")
 * - XP honesty: capture the pre-complete gamification snapshot (whatever is
 *   already cached — honestly, no fabricated baseline), call the complete
 *   endpoint, force-refetch gamification stats, and diff. Only a confirmed
 *   POSITIVE delta is surfaced.
 * - onSettled invalidates BOTH queryKeys.missions() and
 *   queryKeys.gamification() (ADR-4).
 */
export function useCompleteMission() {
  const queryClient = useQueryClient();

  return useMutation<CompleteMissionResult, ApiError, CompleteMissionInput, MissionsSnapshot>({
    mutationFn: async (input) => {
      // "capture stats before" — read whatever is already cached, honestly.
      const preStats = queryClient.getQueryData<GamificationStatsResponseDto>(queryKeys.gamification());

      const response = input.source === 'life'
        ? await completeLifeMission(input.id)
        : await completeHealthMission(input.id);

      // "refetch after" — forced-fresh network call.
      const postStats = await fetchGamificationStats(queryClient);

      const preTotal = preStats?.stats.totalXP;
      const postTotal = postStats.stats.totalXP;
      const rawDelta = typeof preTotal === 'number' ? postTotal - preTotal : null;
      const xpDelta = rawDelta !== null && rawDelta > 0 ? rawDelta : null;

      return { goal: response.goal, xpDelta };
    },
    onMutate: async (input) => {
      await queryClient.cancelQueries({ queryKey: queryKeys.missions() });

      const previousData = queryClient.getQueryData<UnifiedGoalsResponseDto>(queryKeys.missions());

      if (previousData) {
        const optimisticGoals: UnifiedGoalDto[] = previousData.goals.map((goal) =>
          goal.id === input.id ? { ...goal, status: 'completed' as const, progress: 100 } : goal
        );
        queryClient.setQueryData<UnifiedGoalsResponseDto>(queryKeys.missions(), {
          ...previousData,
          goals: optimisticGoals,
        });
      }

      return { previousData };
    },
    onError: (_error, _input, context) => {
      if (context?.previousData) {
        queryClient.setQueryData<UnifiedGoalsResponseDto>(queryKeys.missions(), context.previousData);
      }
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: queryKeys.missions() });
      queryClient.invalidateQueries({ queryKey: queryKeys.gamification() });
    },
  });
}

/**
 * Convenience combined hook for screen packets (S15 create, S14/S42
 * complete+celebration — later waves) that want both actions from one
 * import, in the same `run`/`running`/`error` shape as `useDataExport` /
 * `useDeleteAccount` in `src/hooks/use-trust.ts`.
 */
export function useMissionMutations() {
  const createMutation = useCreateMission();
  const completeMutation = useCompleteMission();

  return {
    createMission: (input: CreateMissionInput) => createMutation.mutateAsync(input),
    creating: createMutation.isPending,
    createError: createMutation.isError ? createMutation.error : null,

    completeMission: (input: CompleteMissionInput) => completeMutation.mutateAsync(input),
    completing: completeMutation.isPending,
    completeError: completeMutation.isError ? completeMutation.error : null,
  };
}
```

## Design spec (verbatim excerpts governing this packet)

This is a data-layer-only packet (no rendering), so the governing spec is the
honesty invariant and the ADR-4 decisions, not visual tokens.

**`Balencia-New-Screens/canon/COMPACT-CANON.md` §7 (verbatim):**
> Every metric ships 3 states:
> 1. **Real** — value + provenance chip (`via WHOOP`, `you logged`).
> 2. **Low-confidence** — muted value + `estimated · low confidence` label.
> 3. **Honest null** — designed empty state, e.g. `Not enough data yet — 3 more days`. Never invent a number.

**Architecture plan ADR-4 (verbatim, `evidence/architecture-plan.md:44-52`):**
> **Decision — tiers.** Extend the missions adapter from 3→6 RPG tiers using only real server signals (`isPrimary`, `source`, `cadence`): `isPrimary && source==='life'` → **life (gold)**; `isPrimary` (other source) → **main (silver)**; `cadence==='weekly'` → **weekly (steel)**; `cadence==='daily'` → **daily (sage)**; default → **side (bronze)**. **copper/group is deferred** (no server squad/group signal) — documented, not faked.
> **Decision — XP honesty.** Completion responses carry **no** XP (server-confirmed). After a successful complete: **refetch `GET /gamification/stats`**, diff `totalXP`/`currentLevel` against a pre-complete snapshot; **celebrate ... only on a confirmed positive delta**. ... never a fabricated toast.

**Amendment A6 (verbatim, `evidence/architecture-plan.md:265`):**
> client `VALID_CATEGORIES` pinned by a unit test against the server list (documented drift tripwire)
