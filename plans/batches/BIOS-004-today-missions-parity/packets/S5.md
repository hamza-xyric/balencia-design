**`celebration-logic.ts` (pure, binding implementation):**

```typescript
// Pure, framework-free parsing for the S42 celebration route's search params
// (BIOS-004 ADR-4 XP honesty + this packet's DEFENSIVE HONESTY requirement).
//
// The S3 mission-completion mutation (`use-mission-mutations.ts`) pushes this
// route ONLY after a REFETCHED gamification-stats snapshot confirms a
// positive XP delta (`CompleteMissionResult.xpDelta: number | null`,
// stringified for the router.push params). This module still treats both
// params as untrusted at the boundary — expo-router's `useLocalSearchParams`
// always returns `string | string[] | undefined` at runtime regardless of
// the TS generic passed to it (a deep link with a repeated query key, e.g.
// `balencia://celebration?xpDelta=1&xpDelta=2`, produces an array even
// though the app's own navigation never does) — so a garbled or absent
// param must resolve to the honest-null branch, never a fabricated number.

export function parseXpDelta(param: string | string[] | undefined): number | null {
  const raw = Array.isArray(param) ? param[0] : param;
  if (raw === undefined) {
    return null;
  }
  const trimmed = raw.trim();
  // Server-confirmed XP deltas are always positive integers (ADR-4) — accept
  // ONLY a plain digit string. No leading '+', no decimals, no scientific
  // notation, no embedded whitespace: any of those are garbled input, not
  // coerced.
  if (!/^\d+$/.test(trimmed)) {
    return null;
  }
  const value = Number(trimmed);
  if (!Number.isFinite(value) || value <= 0) {
    return null;
  }
  return value;
}

const TITLE_MAX_LENGTH = 80;

export function titleForDisplay(param: string | string[] | undefined): string | null {
  const raw = Array.isArray(param) ? param[0] : param;
  if (raw === undefined) {
    return null;
  }
  const trimmed = raw.trim();
  if (trimmed.length === 0) {
    return null;
  }
  if (trimmed.length <= TITLE_MAX_LENGTH) {
    return trimmed;
  }
  return `${trimmed.slice(0, TITLE_MAX_LENGTH - 3)}...`;
}
```

function CelebrationBadge({ reduceMotion }: { reduceMotion: boolean }): JSX.Element {
  const scale = useSharedValue(reduceMotion ? 1 : 0);

  useEffect(() => {
    if (reduceMotion) {
      scale.value = 1;
      return;

    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Medium).catch(() => undefined);

    if (reduceMotion) {
      translateY.value = 0;
      opacity.value = 1;
      return;
    }
    const easing = Easing.bezier(...MotionEasing.physical);
    translateY.value = withTiming(0, { duration: MotionDuration.toastIn, easing });
    opacity.value = withTiming(1, { duration: MotionDuration.toastIn, easing });
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [reduceMotion]);

  const goBack = () => {
    router.back();
  };

  const handleContinue = () => {
    Haptics.impactAsync(Haptics.ImpactFeedbackStyle.Light).catch(() => undefined);

    if (reduceMotion) {
      goBack();
      return;
    }
    const easing = Easing.bezier(...MotionEasing.physical);
    translateY.value = withTiming(24, { duration: MotionDuration.toastOut, easing });
    opacity.value = withTiming(0, { duration: MotionDuration.toastOut, easing }, (finished) => {
      if (finished) {
        runOnJS(goBack)();
      }
    });
  };

  const containerAnimatedStyle = useAnimatedStyle(() => ({
    opacity: opacity.value,
    transform: [{ translateY: translateY.value }],
  }));

  return (
    <View style={styles.scrim} testID="celebration-screen">
      <SafeAreaView edges={['top', 'bottom']} style={styles.safeArea}>
        <Animated.View style={[styles.cardWrapper, containerAnimatedStyle]}>
          <FrostCard glow="done" breathe breatheConfig={CELEBRATION_BREATHE} testID="celebration-card">
            <View style={styles.cardContent}>
              <View style={styles.textBlock}>
                <Text style={styles.overline}>Mission complete</Text>
                {missionTitle !== null && <Text style={styles.subtitle}>{missionTitle}</Text>}
              </View>

              <CelebrationBadge reduceMotion={reduceMotion} />

              {xpDelta !== null ? (
                <View
                  style={styles.xpBlock}
                  accessible
                  accessibilityLabel={`You earned ${xpDelta} XP`}
                >
                  <Text style={styles.xpDisplay}>{`+${xpDelta}`}</Text>
                  <Text style={styles.xpSubcopy}>XP</Text>
                  <ProvenanceChip
                    provenance={{ kind: 'real', source: 'computed', label: 'you earned it' }}
                  />
                </View>
              ) : (
                <Text style={styles.honestBody}>XP confirms within about 15 minutes</Text>
              )}

              <View style={styles.divider} />

              <CIAInsightCard
                headline={null}
                gated={{ reason: 'Cia insight for mission completions arrives in a future update' }}
                testID="celebration-cia-insight"
              />

              <BalenciaButton onPress={handleContinue} accessibilityLabel="Continue">
                Continue
              </BalenciaButton>
            </View>
          </FrostCard>
        </Animated.View>
      </SafeAreaView>
    </View>
  );
}
