import type { Provenance } from '@/components/balencia/states';

export type Metric<T = number> = {
  value: T | null;
  display: string;
  provenance: Provenance;
};

type RealSource = Extract<Provenance, { kind: 'real' }>['source'];

export function realMetric<T>(
  value: T,
  opts: {
    source: RealSource;
    label?: string;
    format?: (v: T) => string;
  },
): Metric<T> {
  const format = opts.format ?? ((v: T) => String(v));
  const label = opts.label ?? 'Real data';
  return {
    value,
    display: format(value),
    provenance: { kind: 'real', source: opts.source, label },
  };
}

export function nullMetric<T = number>(label?: string): Metric<T> {
  return {
    value: null,
    display: '—',
    provenance: { kind: 'null', label: label ?? 'Not enough data yet' },
  };
}

export function lowMetric<T>(
  value: T,
  label: string,
  format?: (v: T) => string,
): Metric<T> {
  const fmt = format ?? ((v: T) => String(v));
  return {
    value,
    display: fmt(value),
    provenance: { kind: 'low', label },
  };
}

export function fromNullable<T>(
  value: T | null | undefined,
  opts: {
    source: RealSource;
    label?: string;
    nullLabel?: string;
    format?: (v: T) => string;
  },
): Metric<T> {
  if (value === null || value === undefined) {
    return nullMetric<T>(opts.nullLabel);
  }
  return realMetric(value, {
    source: opts.source,
    label: opts.label,
    format: opts.format,
  });
}

import type { OverviewDashboardResponse, PulseMetricDto } from '@/services/api/dto/overview';
import { fromNullable, nullMetric, realMetric, type Metric } from '@/services/adapters/provenance';

export type TodayVM = {
  lifePower: Metric<number>;
  pulse: {
    score: Metric<number>;
    metrics: Array<{
      key: 'energy' | 'recovery' | 'stress';
      metric: Metric<number>;
      caption: string | null;
      qualityLabel: string | null;
    }>;
    tip: string | null;
    hasWearable: boolean;
  };
  dailyBrief: {
    headline: string | null;
    opportunity: string | null;
    risk: string | null;
    focusAreas: string[];
  };
  priorities: Array<{
    id: string;
    action: string;
    pillar: string;
    priority: 'high' | 'medium' | 'low';
  }>;
  momentum: {
    streak: Metric<number>;
    xp: Metric<number>;
    level: Metric<number>;
    levelName: string | null;
  };
  domains: Array<{
    key: string;
    label: string;
    score: Metric<number>;
    status: 'strong' | 'steady' | 'attention' | 'no_data';
  }>;
  coachPrompts: string[];
  meta: {
    generatedAt: string;
    hasLifeAreas: boolean;
    hasGoals: boolean;
  };
};

const pulseMetricLabels: Record<Exclude<PulseMetricDto['source'], null>, string> = {
  whoop: 'via WHOOP',
  self_report: 'you logged',
  computed: 'computed',
};

function mapPulseMetric(metric: PulseMetricDto, hasWearable: boolean): Metric<number> {
  // Enforce the wearable requirement for physiological metrics
  if (!hasWearable && (metric.key === 'recovery' || metric.key === 'stress')) {
    return nullMetric('Connect a wearable');
  }

  // Respect server null values (Honesty Invariant)
  if (metric.value === null || metric.source === null) {
    return nullMetric();
  }

  return realMetric(metric.value, {
    source: metric.source,
    label: pulseMetricLabels[metric.source],
    format: () => metric.display ?? String(metric.value),
  });
}

export function toTodayView(dto: OverviewDashboardResponse): TodayVM {
  const { lifeScore, pulse, dailyBrief, priorities, momentum, domains, meta } = dto;

  const pulseMetrics = (pulse?.metrics ?? []).map((m) => ({
    key: m.key as 'energy' | 'recovery' | 'stress',
    metric: mapPulseMetric(m, pulse?.hasWearable ?? false),
    caption: m.caption ?? null,
    qualityLabel: m.qualityLabel ?? null,
  }));

  const vmPriorities = (priorities ?? []).map((p) => ({
    id: p.id,
    action: p.action,
    pillar: p.pillar ?? 'general',
    priority: (p.priority ?? 'medium') as 'high' | 'medium' | 'low',
  }));

  const vmDomains = (domains ?? []).map((d) => ({
    key: d.id,
    label: d.label,
    score: fromNullable(d.score, {
      source: 'server',
      label: 'computed',
      nullLabel: 'No data yet — start tracking',
    }),
    status: (d.status ?? 'no_data') as 'strong' | 'steady' | 'attention' | 'no_data',
  }));

  return {
    lifePower: fromNullable(lifeScore?.totalScore, {
      source: 'computed',
      label: 'computed',
    }),
    pulse: {
      score: fromNullable(pulse?.score, { source: 'computed', label: 'computed' }),
      metrics: pulseMetrics,
      tip: pulse?.tip ?? null,
      hasWearable: pulse?.hasWearable ?? false,
    },
    dailyBrief: {
      headline: dailyBrief?.headline ?? null,
      opportunity: dailyBrief?.opportunity ?? null,
      risk: dailyBrief?.risk ?? null,
      focusAreas: dailyBrief?.focusAreas ?? [],
    },
    priorities: vmPriorities,
    momentum: {
      streak: fromNullable(momentum?.streak, { source: 'server', label: 'from your server' }),
      xp: fromNullable(momentum?.xp, { source: 'server', label: 'from your server' }),
      level: fromNullable(momentum?.level, { source: 'server', label: 'from your server' }),
      levelName: momentum?.levelName ?? null,
    },
    domains: vmDomains,
    coachPrompts: dto.coachPrompts ?? [],
    meta: {
      generatedAt: meta?.generatedAt ?? new Date(0).toISOString(),
      hasLifeAreas: meta?.hasLifeAreas ?? false,
      hasGoals: meta?.hasGoals ?? false,
    },
  };
}

import type { UnifiedGoalsResponseDto, GoalDto, GoalSource } from '@/services/api/dto/goals';

export type MissionTier = 'life' | 'main' | 'side';

export type MissionVM = {
  id: string;
  title: string;
  description: string | null;
  tier: MissionTier;
  status: 'active' | 'blocked' | 'complete';
  progress: number;
  domain: string;
  source: GoalSource;
  isPrimary: boolean;
  dueDate: string | null;
  target: { current: number; target: number; unit: string } | null;
};

export type MissionsVM = {
  missions: MissionVM[];
  summary: {
    total: number;
    active: number;
    completed: number;
    avgProgress: number;
  };
};

function mapTier(goal: GoalDto): MissionTier {
  if (goal.isPrimary && goal.source === 'life') {
    return 'life';
  }
  if (goal.isPrimary) {
    return 'main';
  }
  return 'side';
}

function mapStatus(goal: GoalDto): MissionVM['status'] {
  switch (goal.status) {
    case 'active':
    case 'in_progress':
      return 'active';
    case 'paused':
      return 'blocked';
    case 'completed':
      return 'complete';
    default:
      return 'active';
  }
}

export function toMissionsView(dto: UnifiedGoalsResponseDto): MissionsVM {
  const goals = dto.goals ?? [];
  
  const missions: MissionVM[] = goals.map((goal) => {
    const targetValue = goal.targetValue ?? 0;
    const currentValue = goal.currentValue ?? 0;
    const unit = goal.unit ?? '';
    
    return {
      id: goal.id,
      title: goal.title,
      description: goal.description ?? null,
      tier: mapTier(goal),
      status: mapStatus(goal),
      progress: goal.progress ?? 0,
      domain: goal.domain ?? 'general',
      source: goal.source,
      isPrimary: goal.isPrimary ?? false,
      dueDate: goal.dueDate ?? null,
      target: goal.targetValue !== undefined && goal.targetValue !== null
        ? { current: currentValue, target: targetValue, unit }
        : null,
    };
  });

  const summary = dto.summary ?? {
    total: 0,
    active: 0,
    inProgress: 0,
    paused: 0,
    completed: 0,
    avgProgress: 0,
    bySource: {},
  };

  return {
    missions,
    summary: {
      total: summary.total ?? 0,
      active: (summary.active ?? 0) + (summary.inProgress ?? 0),
      completed: summary.completed ?? 0,
      avgProgress: summary.avgProgress ?? 0,
    },
  };
}

import type { WorkoutPlansResponseDto, WorkoutPlanDto } from '@/services/api/dto/workouts';
import { fromNullable, nullMetric, type Metric } from '@/services/adapters/provenance';

export type FitnessVM = {
  planCount: number;
  activePlan: {
    id: string;
    name: string;
    currentWeek: number;
    durationWeeks: number;
    workoutsPerWeek: number;
    completionRate: Metric<number>;
    totalWorkoutsCompleted: number;
  } | null;
  score: Metric<number>;
  recovery: Metric<number>;
  weeklyMinutes: Metric<number>;
};

export function toFitnessView(dto: WorkoutPlansResponseDto): FitnessVM {
  const plans = dto.plans ?? [];
  const rawActivePlan = plans.find((p) => p.status === 'active') ?? null;

  let activePlan: FitnessVM['activePlan'] = null;
  if (rawActivePlan) {
    activePlan = {
      id: rawActivePlan.id,
      name: rawActivePlan.name,
      currentWeek: rawActivePlan.currentWeek ?? 0,
      durationWeeks: rawActivePlan.durationWeeks ?? 0,
      workoutsPerWeek: rawActivePlan.workoutsPerWeek ?? 0,
      completionRate: fromNullable(rawActivePlan.overallCompletionRate, {
        source: 'server',
        label: 'from your plan',
      }),
      totalWorkoutsCompleted: rawActivePlan.totalWorkoutsCompleted ?? 0,
    };
  }

  return {
    planCount: plans.length,
    activePlan,
    score: nullMetric('No fitness score yet'),
    recovery: nullMetric('Connect a wearable'),
    weeklyMinutes: nullMetric('No workout logs yet'),
  };
}
