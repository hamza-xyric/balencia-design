import { useEffect } from 'react';
import type { JSX } from 'react';
import { StyleSheet, Text, View } from 'react-native';
import Animated, {
  Easing,
  useAnimatedStyle,
  useSharedValue,
  withTiming,
} from 'react-native-reanimated';

import { BalenciaColors, Radius, Spacing } from '@/constants/theme';
import { MotionDuration, MotionEasing } from '@/constants/motion';
import { useReducedMotion } from '@/components/balencia/auth-inputs';

export type MomentumHonesty = 'real' | 'low' | 'null';

export interface MomentumBarProps {
  value: number;
  max?: number;
  state?: MomentumHonesty;
  label?: string;
  glow?: 'you' | 'done';
  testID?: string;
}

export function MomentumBar({
  value,
  max = 100,
  state = 'real',
  label,
  glow = 'you',
  testID,
}: MomentumBarProps): JSX.Element {
  const reduceMotion = useReducedMotion();
  const progress = useSharedValue(0);

  const percent = Math.max(0, Math.min(100, max > 0 ? (value / max) * 100 : 0));

  useEffect(() => {
    if (reduceMotion) {
      progress.value = percent;
    } else {
      progress.value = withTiming(percent, {
        duration: MotionDuration.bandFill,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }
  }, [percent, reduceMotion, progress]);

  const isNull = state === 'null';
  const isLow = state === 'low';

  const animatedFillStyle = useAnimatedStyle(() => ({
    transform: [{ scaleX: progress.value / 100 }],
  }));

  const fillColor = glow === 'done' ? BalenciaColors.green : BalenciaColors.orange;
  const contentOpacity = isLow ? 0.64 : 1;

  const labelTextStyle = [
    styles.label,
    { opacity: contentOpacity },
    isNull && styles.labelNull,
  ];

  return (
    <View style={styles.container} testID={testID}>
      {label !== undefined && (
        <Text style={labelTextStyle}>{label}</Text>
      )}
      <View style={styles.track}>
        {!isNull && (
          <Animated.View
            style={[
              styles.fillBase,
              { backgroundColor: fillColor, opacity: contentOpacity, shadowColor: fillColor },
              animatedFillStyle,
            ]}
          />
        )}
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    gap: Spacing.one,
  },
  label: {
    color: BalenciaColors.paper100,
    fontSize: 13,
    fontWeight: '700',
    fontVariant: ['tabular-nums'],
  },
  labelNull: {
    color: BalenciaColors.paper70,
    fontWeight: '400',
  },
  track: {
    height: 10,
    borderRadius: Radius.sm,
    backgroundColor: 'rgba(254, 250, 243, 0.08)',
    overflow: 'hidden',
  },
  fillBase: {
    position: 'absolute',
    top: 0,
    bottom: 0,
    left: 0,
    width: '100%',
    transformOrigin: 'left',
    borderRadius: Radius.sm,
    shadowOpacity: 0.4,
    shadowOffset: { width: 0, height: 0 },
    shadowRadius: 8,
  },
});
