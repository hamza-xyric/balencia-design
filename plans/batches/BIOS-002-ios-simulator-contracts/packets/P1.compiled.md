```ts
// ==== src/services/api/client.ts (current content at dispatch) ====
import { getAccessToken } from '@/services/auth/session';
import { Env } from '@/config/env';

type ApiEnvelope<T> = {
  success?: boolean;
  message?: string;
  data?: T;
  error?: string;
  errors?: unknown[];
};

type ApiFetchOptions = Omit<RequestInit, 'body'> & {
  body?: unknown;
  auth?: boolean;
};

export const API_BASE_URL = Env.apiUrl;

export class ApiError extends Error {
  status: number;
  details?: unknown;

  constructor(message: string, status: number, details?: unknown) {
    super(message);
    this.name = 'ApiError';
    this.status = status;
    this.details = details;
  }
}

export async function apiFetch<T>(path: string, options: ApiFetchOptions = {}): Promise<T> {
  const { auth = true, body, headers, ...requestOptions } = options;
  const token = auth ? await getAccessToken() : null;
  const normalizedPath = path.startsWith('/') ? path : `/${path}`;
  const baseHeaders: Record<string, string> = {
    Accept: 'application/json',
    'Content-Type': 'application/json',
    'X-Client': 'mobile',
  };

  if (token) {
    baseHeaders.Authorization = `Bearer ${token}`;
  }

  const response = await fetch(`${API_BASE_URL}${normalizedPath}`, {
    ...requestOptions,
    headers: {
      ...baseHeaders,
      ...(headers as Record<string, string> | undefined),
    },
    body: body === undefined ? undefined : JSON.stringify(body),
  });

  const text = await response.text();
  const parsed = text ? parseJson(text) : null;

  if (!response.ok) {
    const envelope = parsed as ApiEnvelope<unknown> | null;
    throw new ApiError(
      envelope?.message ?? envelope?.error ?? `Request failed with ${response.status}`,
      response.status,
      envelope?.errors ?? envelope
    );
  }

  const envelope = parsed as ApiEnvelope<T> | T | null;
  if (isEnvelope(envelope)) {
    if (envelope.success === false) {
      throw new ApiError(envelope.message ?? envelope.error ?? 'Request failed', response.status, envelope.errors);
    }
    return envelope.data as T;
  }

  return envelope as T;
}

function parseJson(text: string): unknown {
  try {
    return JSON.parse(text);
  } catch {
    return text;
  }
}

function isEnvelope<T>(value: ApiEnvelope<T> | T | null): value is ApiEnvelope<T> {
  return Boolean(value && typeof value === 'object' && ('success' in value || 'data' in value));
}

```

## Contracts (implement exactly)

```ts
export class ApiError extends Error {
  readonly status: number;          // HTTP status (0 for network failure)
  readonly code?: string;           // server ApiErrorBody.code e.g. 'UNAUTHORIZED', 'FEATURE_DISABLED'
  readonly requestId?: string;
  readonly details?: unknown;
  constructor(args: { message: string; status: number; code?: string; requestId?: string; details?: unknown });
}
export class SessionExpiredError extends ApiError {}   // thrown when refresh fails / no session
export class NetworkError extends ApiError {}          // fetch threw (offline etc.), status 0
```

```ts
  method?: 'GET'|'POST'|'PATCH'|'PUT'|'DELETE';
  body?: unknown;                    // JSON.stringify'd when present
  headers?: Record<string,string>;   // merged over defaults
  skipAuth?: boolean;                // true for login/refresh (no Bearer, no staleness check)
  retry?: boolean;                   // default true for GET, false otherwise
  timeoutMs?: number;                // default 15000, AbortController-based
};
export async function apiFetch<T>(path: string, options?: ApiFetchOptions): Promise<T>;

// Session integration (P3 will call these at app start; keep module-level holders):

```

Behavior requirements:
1. URL = `${env.apiUrl}${path}` (path always starts with `/`). Default headers: `Content-Type: application/json`, `X-Client: mobile`; plus `Authorization: Bearer <token>` when `!skipAuth` and provider returns a token.
2. Before an authed request, await the stale guard if set (it may refresh proactively).
3. Envelope handling: parse JSON; if `success === true` return `data` as `T`; if `success === false` throw `ApiError` with server `message`/`code`/`requestId` and the HTTP status. Non-JSON response → `ApiError` with status and generic message.
4. **401 handling (reactive):** on HTTP 401 for an authed request, call the refresh handler ONCE; if it resolves `true`, re-read the token and retry the original request exactly once; a second 401 (or refresh `false`/absent) throws `SessionExpiredError`. Never loop.
5. **Retry policy:** network failures and 5xx retry up to 3 attempts total with delays 1000ms → 2000ms → 4000ms, ONLY when `retry` resolves true (default: GET yes, mutations no). 4xx never retries. 401-refresh-retry is separate from and not counted by this policy.
6. `timeoutMs` via `AbortController`; abort → `NetworkError` (status 0, message mentions timeout).
7. Concurrency: if multiple requests 401 simultaneously, the refresh handler must be invoked only once — share one in-flight refresh promise inside the client.
8. No `console.log`. No imports from session/auth modules (setters only — import cycle ban).
