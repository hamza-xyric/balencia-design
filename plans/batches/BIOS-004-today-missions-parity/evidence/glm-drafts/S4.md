import { VALID_LIFE_GOAL_CATEGORIES } from '@/services/api/dto/goals';
import type { CreateLifeGoalRequestDto, LifeGoalCategory } from '@/services/api/dto/goals';

export interface CreateMissionValidation {
  valid: boolean;
  reason?: 'empty-title' | 'invalid-category';
}

export function validateCreateInput(input: { title: string; category: string }): CreateMissionValidation {
  const trimmedTitle = input.title.trim();
  if (trimmedTitle.length === 0) {
    return { valid: false, reason: 'empty-title' };
  }
  if (!VALID_LIFE_GOAL_CATEGORIES.includes(input.category as LifeGoalCategory)) {
    return { valid: false, reason: 'invalid-category' };
  }
  return { valid: true };
}

export function categoryLabel(category: LifeGoalCategory): string {
  const spaced = category.replace(/_/g, ' ');
  return spaced.charAt(0).toUpperCase() + spaced.slice(1);
}

export interface BuildCreateBodyInput {
  title: string;
  category: LifeGoalCategory;
  isPrimary?: boolean;
}

export function buildCreateBody(input: BuildCreateBodyInput): CreateLifeGoalRequestDto {
  const body: CreateLifeGoalRequestDto = {
    title: input.title.trim(),
    category: input.category,
  };
  if (input.isPrimary !== undefined) {
    body.is_primary = input.isPrimary;
  }
  return body;
}

import { useEffect, useState, type JSX } from 'react';
import { Animated, Easing, KeyboardAvoidingView, Platform, Pressable, ScrollView, StyleSheet, Text, View } from 'react-native';
import { SafeAreaView } from 'react-native-safe-area-context';
import { useRouter } from 'expo-router';
import * as Haptics from 'expo-haptics';

  BalenciaButton,
  ChipDomainTag,
  GlassPillInput,
  SegmentedTabs,
  ToastBanner,
  useReducedMotion,
  type SegmentedTabsOption,
} from '@/components/balencia';
import { BalenciaColors, Radius, Spacing, TouchTarget } from '@/constants/theme';
import { MotionDuration, MotionEasing } from '@/constants/motion';
import { useCreateMission } from '@/hooks/use-mission-mutations';
import { ApiError } from '@/services/api/errors';
import { VALID_LIFE_GOAL_CATEGORIES, type LifeGoalCategory } from '@/services/api/dto/goals';
import { validateCreateInput, categoryLabel, buildCreateBody } from '@/features/missions/mission-create-logic';

type MissionTypeKey = 'main' | 'side';

const MISSION_TYPE_OPTIONS: SegmentedTabsOption<MissionTypeKey>[] = [
  { key: 'main', label: 'Main mission' },
  { key: 'side', label: 'Side mission' },
];

export function MissionCreateScreen(): JSX.Element {
  const router = useRouter();
  const reduceMotion = useReducedMotion();
  const createMutation = useCreateMission();

  const [title, setTitle] = useState('');
  const [category, setCategory] = useState<LifeGoalCategory | ''>('');
  const [missionType, setMissionType] = useState<MissionTypeKey>('side');
  const [toast, setToast] = useState<string | null>(null);

  const [inputScale] = useState(() => new Animated.Value(1));
  const [successOpacity] = useState(() => new Animated.Value(0));

  const validation = validateCreateInput({ title, category });

  useEffect(() => {
    if (reduceMotion) {
      inputScale.setValue(1);
      return;
    }
    Animated.timing(inputScale, {
      toValue: createMutation.isPending ? 0.97 : 1,
      duration: MotionDuration.processingShrink,
      easing: Easing.bezier(...MotionEasing.physical),
      useNativeDriver: true,
    }).start();
  }, [createMutation.isPending, reduceMotion, inputScale]);

  async function handleSubmit() {
    if (!validation.valid || createMutation.isPending) return;
    setToast(null);
    try {
      await createMutation.mutateAsync(
        buildCreateBody({
          title,
          category: category as LifeGoalCategory,
          isPrimary: missionType === 'main',
        })
      );
      Haptics.notificationAsync(Haptics.NotificationFeedbackType.Success).catch(() => undefined);
      if (reduceMotion) {
        router.back();
        return;
      }
      Animated.timing(successOpacity, {
        toValue: 1,
        duration: MotionDuration.successSwap,
        easing: Easing.bezier(...MotionEasing.physical),
        useNativeDriver: true,
      }).start(() => {
        router.back();
      });
    } catch (submitError) {
      if (submitError instanceof ApiError) {
        setToast(submitError.message);
      } else {
        setToast('Something went wrong. Try again.');
      }
    }
  }

  return (
    <KeyboardAvoidingView behavior={Platform.select({ ios: 'padding', default: undefined })} style={styles.flex}>
      <SafeAreaView edges={['top', 'bottom', 'left', 'right']} style={styles.safeArea}>
        <View style={styles.header}>
          <View style={styles.headerSpacer} />
          <Text style={styles.headerTitle}>New mission</Text>
          <Pressable
            onPress={() => router.back()}
            accessibilityRole="button"
            accessibilityLabel="Close"
            style={styles.closeButton}
            testID="mission-create-close">
            <Text style={styles.closeGlyph}>✕</Text>
          </Pressable>
        </View>

        <ScrollView
          style={styles.scroll}
          contentContainerStyle={styles.content}
          keyboardShouldPersistTaps="handled"
          showsVerticalScrollIndicator={false}>
          {toast ? <ToastBanner message={toast} tone="error" onDismiss={() => setToast(null)} /> : null}

          <Animated.View style={{ transform: [{ scale: inputScale }] }}>
            <GlassPillInput
              label="Title"
              value={title}
              onChangeText={setTitle}
              variant="text"
              placeholder="What do you want to achieve"
              disabled={createMutation.isPending}
              testID="mission-create-title"
            />

            <View style={styles.section}>
              <Text style={styles.sectionLabel}>Category</Text>
              <View style={styles.chipGrid}>
                {VALID_LIFE_GOAL_CATEGORIES.map((cat) => (
                  <ChipDomainTag
                    key={cat}
                    label={categoryLabel(cat)}
                    active={category === cat}
                    onPress={createMutation.isPending ? undefined : () => setCategory(cat)}
                    testID={`mission-create-category-${cat}`}
                  />
                ))}
              </View>
            </View>

            <View style={styles.section}>
              <Text style={styles.sectionLabel}>Mission type</Text>
              <SegmentedTabs
                options={MISSION_TYPE_OPTIONS}
                value={missionType}
                onChange={(key) => {
                  if (!createMutation.isPending) setMissionType(key);
                }}
                accessibilityLabel="Mission type"
                testID="mission-create-type"
              />
            </View>
          </Animated.View>
        </ScrollView>

        <View style={styles.footer}>
          <View testID="mission-create-submit" style={styles.buttonWrap}>
            <BalenciaButton
              disabled={!validation.valid || createMutation.isPending}
              onPress={handleSubmit}
              accessibilityLabel="Create mission">
              {createMutation.isPending ? 'Creating mission' : 'Create mission'}
            </BalenciaButton>
            <Animated.View
              pointerEvents="none"
              style={[styles.successOverlay, { opacity: reduceMotion ? 0 : successOpacity }]}
            />
          </View>
        </View>
      </SafeAreaView>
    </KeyboardAvoidingView>
  );
}

import { describe, it, expect } from 'vitest';
import {
  validateCreateInput,
  categoryLabel,
  buildCreateBody,
} from '@/features/missions/mission-create-logic';
import { VALID_LIFE_GOAL_CATEGORIES } from '@/services/api/dto/goals';

const SERVER_VALID_LIFE_GOAL_CATEGORIES = [
  'spiritual', 'social', 'productivity', 'happiness',
  'anxiety_management', 'creative', 'personal_growth',
  'financial', 'faith', 'relationships', 'education',
  'career', 'health_wellness', 'custom',
] as const;

describe('validateCreateInput', () => {
  it('returns valid for a valid title and category', () => {
    expect(validateCreateInput({ title: 'Run a 5K', category: 'spiritual' })).toEqual({ valid: true });
  });

  it('returns empty-title for an empty title', () => {
    expect(validateCreateInput({ title: '', category: 'spiritual' })).toEqual({ valid: false, reason: 'empty-title' });
  });

  it('returns empty-title for a whitespace-only title', () => {
    expect(validateCreateInput({ title: '   ', category: 'spiritual' })).toEqual({ valid: false, reason: 'empty-title' });
  });

  it('returns invalid-category for a bogus category', () => {
    expect(validateCreateInput({ title: 'Run a 5K', category: 'not-a-real-category' })).toEqual({ valid: false, reason: 'invalid-category' });
  });

  it('returns invalid-category for an empty-string category (initial unselected state)', () => {
    expect(validateCreateInput({ title: 'Run a 5K', category: '' })).toEqual({ valid: false, reason: 'invalid-category' });
  });

  it('prioritizes empty-title over invalid-category', () => {
    expect(validateCreateInput({ title: '', category: 'not-real' })).toEqual({ valid: false, reason: 'empty-title' });
  });
});

describe('categoryLabel', () => {
  it('formats all valid categories correctly', () => {
    const expected: Record<string, string> = {
      spiritual: 'Spiritual',
      social: 'Social',
      productivity: 'Productivity',
      happiness: 'Happiness',
      anxiety_management: 'Anxiety management',
      creative: 'Creative',
      personal_growth: 'Personal growth',
      financial: 'Financial',
      faith: 'Faith',
      relationships: 'Relationships',
      education: 'Education',
      career: 'Career',
      health_wellness: 'Health wellness',
      custom: 'Custom',
    };

    for (const cat of VALID_LIFE_GOAL_CATEGORIES) {
      expect(categoryLabel(cat)).toBe(expected[cat]);
    }
  });
});

describe('buildCreateBody', () => {
  it('trims the title and omits is_primary when isPrimary is undefined', () => {
    const result = buildCreateBody({ title: '  Run a 5K  ', category: 'spiritual' });
    expect(result).toEqual({ title: 'Run a 5K', category: 'spiritual' });
    expect(result).not.toHaveProperty('is_primary');
  });

  it('sets is_primary to true when isPrimary is true', () => {
    const result = buildCreateBody({ title: 'Run a 5K', category: 'spiritual', isPrimary: true });
    expect(result.is_primary).toBe(true);
  });

  it('sets is_primary to false when isPrimary is false', () => {
    const result = buildCreateBody({ title: 'Run a 5K', category: 'spiritual', isPrimary: false });
    expect(result.is_primary).toBe(false);
  });
});
