import { useEffect, useState } from 'react';
import type { JSX } from 'react';
import React from 'react';
import { StyleSheet, Text, View } from 'react-native';
import { Circle, Svg } from 'react-native-svg';
import Animated, {
  Easing,
  useAnimatedProps,
  useSharedValue,
  withTiming,
} from 'react-native-reanimated';

import { BalenciaColors } from '@/constants/theme';
import { MotionDuration, MotionEasing } from '@/constants/motion';
import { useReducedMotion } from '@/components/balencia/auth-inputs';

export type Honesty = 'real' | 'low' | 'null';

export interface ProgressRingProps {
  progress: number;
  size?: number;
  stroke?: number;
  state?: Honesty;
  centerLabel?: string | null;
  glyph?: React.ReactNode;
  glow?: 'you' | 'done' | 'cia';
  animateOnMount?: boolean;
  testID?: string;
}

const AnimatedCircle = Animated.createAnimatedComponent(Circle);

export function ProgressRing({
  progress,
  size = 96,
  stroke = 8,
  state = 'real',
  centerLabel,
  glyph,
  glow = 'you',
  animateOnMount = true,
  testID,
}: ProgressRingProps): JSX.Element {
  const reduceMotion = useReducedMotion();
  const [hasMounted, setHasMounted] = useState(false);
  const clampedProgress = Math.min(100, Math.max(0, progress));

  const dashOffset = useSharedValue(0);

  const radius = (size - stroke) / 2;
  const circumference = 2 * Math.PI * radius;
  const finalOffset = circumference * (1 - clampedProgress / 100);

  useEffect(() => {
    const isFirstMount = !hasMounted;
    
    const shouldSkipAnimation =
      reduceMotion || (isFirstMount && !animateOnMount);

    if (shouldSkipAnimation) {
      dashOffset.value = finalOffset;
    } else {
      const duration = isFirstMount
        ? MotionDuration.sweep
        : MotionDuration.sweepShort;

      dashOffset.value = withTiming(finalOffset, {
        duration,
        easing: Easing.bezier(...MotionEasing.physical),
      });
    }

    if (isFirstMount) {
      setHasMounted(true);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [clampedProgress, reduceMotion, animateOnMount]);

  const arcColor = (() => {
    if (state === 'null') return 'transparent';
    if (glow === 'cia') return BalenciaColors.purple;
    if (glow === 'done') return BalenciaColors.green;
    return clampedProgress >= 100
      ? BalenciaColors.green
      : BalenciaColors.orange;
  })();

  const isLowConfidence = state === 'low';
  const arcOpacity = state === 'null' ? 0 : isLowConfidence ? 0.64 : 1;

  const animatedProps = useAnimatedProps(() => ({
    strokeDashoffset: dashOffset.value,
  }));

  const showLabel = state !== 'null' && centerLabel !== null && centerLabel !== undefined && centerLabel !== '';
  const showGlyph = !showLabel && glyph !== undefined && glyph !== null;

  const accessibilityLabel = (() => {
    if (state === 'null') return 'Progress unavailable';
    return `Progress ${Math.round(clampedProgress)} percent`;
  })();

  const accessibilityValue =
    state === 'null'
      ? { min: 0, max: 100 }
      : { min: 0, max: 100, now: Math.round(clampedProgress) };

  return (
    <View
      style={[styles.container, { width: size, height: size }]}
      testID={testID}
      accessibilityRole="progressbar"
      accessibilityValue={accessibilityValue}
      accessibilityLabel={accessibilityLabel}
    >
      <View
        style={{
          width: size,
          height: size,
          transform: [{ rotate: '-90deg' }],
        }}
      >
        <Svg width={size} height={size}>
          <Circle
            cx={size / 2}
            cy={size / 2}
            r={radius}
            stroke={BalenciaColors.chartTrack}
            strokeWidth={stroke}
            fill="none"
          />
          {state !== 'null' && (
            <AnimatedCircle
              cx={size / 2}
              cy={size / 2}
              r={radius}
              stroke={arcColor}
              strokeWidth={stroke}
              strokeDasharray={circumference}
              strokeLinecap="round"
              fill="none"
              opacity={arcOpacity}
              animatedProps={animatedProps}
            />
          )}
        </Svg>
      </View>

      {showLabel && (
        <View style={styles.centerContent} pointerEvents="none">
          <Text
            style={[
              styles.label,
              isLowConfidence && styles.labelLowConfidence,
            ]}
          >
            {centerLabel}
          </Text>
        </View>
      )}

      {showGlyph && (
        <View style={styles.centerContent} pointerEvents="none">
          {glyph}
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    justifyContent: 'center',
    alignItems: 'center',
  },
  centerContent: {
    position: 'absolute',
    justifyContent: 'center',
    alignItems: 'center',
  },
  label: {
    color: BalenciaColors.paper100,
    fontFamily: 'system-ui',
    fontWeight: '600',
    fontSize: 18,
    fontVariant: ['tabular-nums'],
  },
  labelLowConfidence: {
    opacity: 0.64,
  },
});
