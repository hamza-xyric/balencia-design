export type ProvenanceKind = 'live' | 'null' | 'unknown';

export interface Provenance {
  kind: ProvenanceKind;
  /** ISO timestamp if available */
  fetchedAt?: string;
  /** Optional human readable source */
  source?: string;
}

export function liveProvenance(fetchedAt?: string): Provenance {
  return { kind: 'live', fetchedAt: fetchedAt ?? new Date().toISOString(), source: 'server' };
}

export function nullProvenance(): Provenance {
  return { kind: 'null', source: 'empty' };
}

export function unknownProvenance(reason?: string): Provenance {
  return { kind: 'unknown', source: reason ?? 'unavailable' };
}

    timeoutMs: 30000,

export async function deleteMyAccount(args: { password?: string }): Promise<DeleteAccountResponseDto> {
  const body: { password?: string; confirmation: typeof DELETE_ACCOUNT_CONFIRMATION } = {
    confirmation: DELETE_ACCOUNT_CONFIRMATION,
  };

  if (typeof args.password === 'string' && args.password.length > 0) {
    body.password = args.password;
  }

  return apiFetch<DeleteAccountResponseDto>('/v1/users/me', {
    method: 'DELETE',
    body,
    retry: false,
  });
}

  return apiFetch<PrivacyPatchResponseDto>('/preferences/privacy', {
    method: 'PATCH',
    body: patch,

import { UserDataExportDto, PrivacyPreferencesDto } from '@/services/api/dto';

export type ExportTableSummaryVM = {
  name: string;
  exported: number;
  total: number;
  truncated: boolean;
};

export type ExportSummaryVM = {
  exportedAt: string;
  tables: ExportTableSummaryVM[];
  anyTruncated: boolean;
  raw: UserDataExportDto;
};

export function toExportSummary(dto: UserDataExportDto): ExportSummaryVM {
  const tables: ExportTableSummaryVM[] = [];
  let anyTruncated = false;

  for (const [name, count] of Object.entries(dto.counts)) {
    const total = typeof count.total === 'number' && Number.isFinite(count.total) ? count.total : 0;
    const exported = typeof count.exported === 'number' && Number.isFinite(count.exported) ? count.exported : 0;
    const truncated = exported < total;
    
    if (truncated) {
      anyTruncated = true;
    }
    
    tables.push({ name, exported, total, truncated });
  }

  return {
    exportedAt: dto.exportedAt,
    tables,
    anyTruncated,
    raw: dto,
  };
}

import { useMemo } from 'react';
import { useMutation, useQuery, useQueryClient } from '@tanstack/react-query';

import { queryKeys } from '@/services/query/keys';
import {
  deleteMyAccount,
  exportMyData,
  getPreferences,
  updatePrivacy,
} from '@/services/api/trust';
import { ApiError } from '@/services/api/errors';
import { toExportSummary, ExportSummaryVM, PrivacyPrefsVM } from '@/services/adapters/trust';
import { PrivacyPreferencesDto, PreferencesResponseDto, PrivacyPatchResponseDto, DeleteAccountResponseDto } from '@/services/api/dto';

export function usePrivacyPrefs(): {
  prefs: PrivacyPrefsVM | null;
  status: 'pending' | 'error' | 'success';
  update: (patch: Partial<PrivacyPrefsVM>) => void;
  updating: boolean;
} {
  const queryClient = useQueryClient();

  const query = useQuery<PreferencesResponseDto, ApiError>({
    queryKey: queryKeys.preferences(),
    queryFn: getPreferences,
  });

  const prefs = useMemo(() => {
    if (!query.data) {
      return null;
    }
    return query.data.preferences.privacy;
  }, [query.data]);

  const mutation = useMutation<PrivacyPatchResponseDto, ApiError, Partial<PrivacyPreferencesDto>>({
    mutationFn: (patch: Partial<PrivacyPreferencesDto>) => updatePrivacy(patch),
    onMutate: async (patch: Partial<PrivacyPreferencesDto>) => {
      await queryClient.cancelQueries({ queryKey: queryKeys.preferences() });

      const previousData = queryClient.getQueryData<PreferencesResponseDto>(queryKeys.preferences());

      if (previousData) {
        const optimisticData: PreferencesResponseDto = {
          ...previousData,
          preferences: {
            ...previousData.preferences,
            privacy: {
              ...previousData.preferences.privacy,
              ...patch,
            },
          },
        };
        queryClient.setQueryData<PreferencesResponseDto>(queryKeys.preferences(), optimisticData);
      }

      return { previousData };
    },
    onError: (_error, _patch, context) => {
      if (context?.previousData) {
        queryClient.setQueryData<PreferencesResponseDto>(queryKeys.preferences(), context.previousData);
      }
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: queryKeys.preferences() });
    },
  });

  return {
    prefs,
    status: query.status,
    update: (patch: Partial<PrivacyPrefsVM>) => mutation.mutate(patch),
    updating: mutation.isPending,
  };
}

export function useDataExport(): {
  run: () => Promise<ExportSummaryVM>;
  running: boolean;
  result: ExportSummaryVM | null;
  error: string | null;
} {
  const mutation = useMutation<ExportSummaryVM, ApiError, void>({
    mutationFn: async () => {
      const dto = await exportMyData();
      return toExportSummary(dto);
    },
  });

  const run = async () => {
    return mutation.mutateAsync();
  };

  return {
    run,
    running: mutation.isPending,
    result: mutation.data ?? null,
    error: mutation.isError ? (mutation.error?.message ?? 'Failed to export data.') : null,
  };
}

export function useDeleteAccount(): {
  run: (args: { password?: string }) => Promise<boolean>;
  running: boolean;
  error: string | null;
} {
  const mutation = useMutation<DeleteAccountResponseDto, ApiError, { password?: string }>({
    mutationFn: async (args: { password?: string }) => deleteMyAccount(args),
  });

  const run = async (args: { password?: string }) => {
    try {
      await mutation.mutateAsync(args);
      return true;
    } catch (error) {
      if (error instanceof ApiError) {
        if (error.status === 401) {
          mutation.reset();
          mutation.setError(new Error('Password is incorrect') as unknown as ApiError);
        } else if (error.status === 400) {
          mutation.reset();
          mutation.setError(new Error(error.message || 'Failed to delete account.') as unknown as ApiError);
        } else {
          mutation.reset();
          mutation.setError(new Error(error.message || 'Failed to delete account.') as unknown as ApiError);
        }
      }
      return false;
    }
  };

  return {
    run,
    running: mutation.isPending,
    error: mutation.isError ? (mutation.error?.message ?? 'Failed to delete account.') : null,
  };
}
