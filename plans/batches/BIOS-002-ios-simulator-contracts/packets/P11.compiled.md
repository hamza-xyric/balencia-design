const starterMessages: CiaMessage[] = [
  {
    id: 'cia-1',
    role: 'cia',
    text: 'I can help tune your Today plan. For this pilot, I will label anything that depends on unverified mobile contracts.',
    confidence: 'medium',
  },
];

export function CiaScreen() {
  const [messages, setMessages] = useState<CiaMessage[]>(starterMessages);
  const [draft, setDraft] = useState('');
  const [isSending, setSending] = useState(false);

  async function sendMessage() {
    const trimmed = draft.trim();
    if (!trimmed) {
      return;
    }

    const userMessage: CiaMessage = {
      id: `user-${Date.now()}`,
      role: 'user',
      text: trimmed,
    };
    const nextMessages = [...messages, userMessage];
    setMessages(nextMessages);
    setDraft('');
    setSending(true);

    try {
      // Interim wiring: P11 replaces this with useCiaChat (optimistic sends + gated states).
      const response = await sendCiaChat({ message: trimmed });
      setMessages([
        ...nextMessages,
        {
          id: `cia-${Date.now()}`,
          role: 'cia',
          text: response.message,
          confidence: 'medium',
        },
      ]);
    } catch {
      setMessages([
        ...nextMessages,
        {
          id: `cia-${Date.now()}`,
          role: 'cia',
          text: 'I can draft the next mission step locally, but live Cia chat needs a signed-in mobile token and entitlement verification.',
          confidence: 'low',
        },
      ]);
    } finally {
      setSending(false);
    }
  }

  return (
    <BalenciaScreen
      eyebrow="Cia"
      title="Coach with provenance."
      subtitle="Every response is either from the backend route or marked as a local pilot fallback.">
      <LegacyProvenanceChip label="Route: /ai-coach/chat, entitlement gated" />

        <GlassCard key={message.id} tone={message.role === 'cia' ? 'purple' : 'default'}>

            <Text style={styles.messageRole}>{message.role === 'cia' ? 'Cia' : 'You'}</Text>
            {message.confidence ? <StatusChip label={`${message.confidence} confidence`} tone={message.confidence === 'low' ? 'warning' : 'muted'} /> : null}
          </View>
          <Text style={styles.messageText}>{message.text}</Text>
        </GlassCard>
      ))}
      <View style={styles.composer}>
        <TextInput
          multiline
          onChangeText={setDraft}
          placeholder="Ask Cia what to focus on next"
          placeholderTextColor={BalenciaColors.paper55}
          style={styles.input}
          value={draft}
        />
        <BalenciaButton disabled={!draft.trim() || isSending} onPress={sendMessage}>
          {isSending ? 'Sending' : 'Send'}
        </BalenciaButton>
      </View>
    </BalenciaScreen>
  );
}

import React, {
  createContext,
  useCallback,
  useContext,
  useEffect,
  useMemo,
  useReducer,
  useRef,
  type ReactNode,
} from 'react';

import {
  apiFetch,
  setAuthTokenProvider,
  setRefreshHandler,
  setStaleGuard,
} from '@/services/api/client';
import { ApiError, SessionExpiredError } from '@/services/api/errors';
import type {
  LoginResponseDto,
  MeResponseDto,
  PublicUserProfileDto,
} from '@/services/api/dto/auth';

import {
  clearSession,
  getAccessToken,
  getStoredSession,
  isTokenStale,
  refreshSession,
  saveSession,
  updateStoredUser,
} from '@/services/auth/session';
import {
  initialSessionState,
  sessionReducer,
  type SessionStatus,
} from '@/services/auth/session-machine';

export type SessionContextValue = {
  status: SessionStatus;
  user: PublicUserProfileDto | null;
  error: string | null;
  signIn(email: string, password: string): Promise<void>;
  signOut(): Promise<void>;
  refreshUser(): Promise<void>;
};

export type { SessionStatus };

const Context = createContext<SessionContextValue | null>(null);

export function SessionProvider(props: {
  children: ReactNode;
  onSignedIn?: () => void;
  onAuthCleared?: () => void;
}): React.JSX.Element {
  const [state, dispatch] = useReducer(sessionReducer, initialSessionState);
  const onSignedInRef = useRef(props.onSignedIn);
  const onAuthClearedRef = useRef(props.onAuthCleared);

  useEffect(() => {
    onSignedInRef.current = props.onSignedIn;
    onAuthClearedRef.current = props.onAuthCleared;
  }, [props.onSignedIn, props.onAuthCleared]);

  const handleAuthCleared = useCallback(() => {
    onAuthClearedRef.current?.();
  }, []);

  const refreshUser = useCallback(async () => {
    try {
      const data = await apiFetch<MeResponseDto>('/auth/me', { method: 'GET' });
      await updateStoredUser(data.user);
      dispatch({ type: 'USER_UPDATED', user: data.user });
    } catch (error) {
      if (error instanceof SessionExpiredError) {
        dispatch({ type: 'REFRESH_FAILURE' });
        handleAuthCleared();
      } else {
        throw error;
      }
    }
  }, [handleAuthCleared]);

  useEffect(() => {
    let mounted = true;

    async function boot() {
      const stored = await getStoredSession();
      if (!mounted) return;

      if (!stored) {
        dispatch({ type: 'NO_SESSION' });
        return;
      }

      // Optimistically hydrate
      dispatch({ type: 'HYDRATED', user: stored.user });

      // Background validate
      try {
        const data = await apiFetch<MeResponseDto>('/auth/me', { method: 'GET' });
        if (!mounted) return;
        await updateStoredUser(data.user);
        dispatch({ type: 'USER_UPDATED', user: data.user });
      } catch (error) {
        if (!mounted) return;
        if (error instanceof SessionExpiredError) {
          dispatch({ type: 'REFRESH_FAILURE' });
          handleAuthCleared();
        }
        // Other errors leave user in optimistically hydrated authenticated state
      }
    }

    boot();

    return () => {
      mounted = false;
    };
  }, [handleAuthCleared]);

  const signIn = useCallback(
    async (email: string, password: string) => {
      dispatch({ type: 'SIGN_IN_START' });
      try {
        const data = await apiFetch<LoginResponseDto>('/auth/login', {
          method: 'POST',
          skipAuth: true,
          body: { email, password },
        });

        await saveSession({ user: data.user, tokens: data.tokens });
        dispatch({ type: 'SIGN_IN_SUCCESS', user: data.user });
        onSignedInRef.current?.();
      } catch (error) {
        const message =
          error instanceof ApiError || error instanceof Error ? error.message : 'Sign in failed';
        dispatch({ type: 'SIGN_IN_FAILURE', error: message });
        throw error;
      }
    },
    []
  );

  const signOut = useCallback(async () => {
    try {
      await apiFetch<null>('/auth/logout', { method: 'POST', retry: false });
    } catch {
      // Swallow errors during best-effort logout
    }

    await clearSession();
    dispatch({ type: 'SIGNED_OUT' });
    handleAuthCleared();
  }, [handleAuthCleared]);

  // Register P1 Client Hooks
  useEffect(() => {
    setAuthTokenProvider(getAccessToken);
    setRefreshHandler(refreshSession);
    setStaleGuard(async () => {
      const s = await getStoredSession();
      if (s && isTokenStale(s)) {
        try {
          await refreshSession();
        } catch {
          // Swallow: if it fails, the API request will naturally 401 and handle it
        }
      }
    });
  }, []);

  const value = useMemo<SessionContextValue>(
    () => ({
      status: state.status,
      user: state.user,
      error: state.error,
      signIn,
      signOut,
      refreshUser,
    }),
    [state, signIn, signOut, refreshUser]
  );

  return <Context.Provider value={value}>{props.children}</Context.Provider>;
}

export function useSession(): SessionContextValue {
  const ctx = useContext(Context);
  if (!ctx) {
    throw new Error('useSession must be used within a SessionProvider');
  }
  return ctx;
}

import type { PublicUserProfileDto } from '@/services/api/dto/auth';

export type SessionStatus =
  | 'booting'
  | 'unauthenticated'
  | 'authenticating'
  | 'authenticated'
  | 'refreshing'
  | 'expired';

export type SessionState = {
  status: SessionStatus;
  user: PublicUserProfileDto | null;
  error: string | null;
};

export type SessionEvent =
  | { type: 'HYDRATED'; user: PublicUserProfileDto }
  | { type: 'NO_SESSION' }
  | { type: 'SIGN_IN_START' }
  | { type: 'SIGN_IN_SUCCESS'; user: PublicUserProfileDto }
  | { type: 'SIGN_IN_FAILURE'; error: string }
  | { type: 'REFRESH_START' }
  | { type: 'REFRESH_SUCCESS' }
  | { type: 'REFRESH_FAILURE' }
  | { type: 'USER_UPDATED'; user: PublicUserProfileDto }
  | { type: 'SIGNED_OUT' };

export const initialSessionState: SessionState = {
  status: 'booting',
  user: null,
  error: null,
};

export function sessionReducer(state: SessionState, event: SessionEvent): SessionState {
  switch (event.type) {
    case 'HYDRATED': {
      return { status: 'authenticated', user: event.user, error: null };
    }
    case 'NO_SESSION': {
      return { status: 'unauthenticated', user: null, error: null };
    }
    case 'SIGN_IN_START': {
      if (state.status === 'authenticated' || state.status === 'authenticating') {
        return state;
      }
      return { status: 'authenticating', user: null, error: null };
    }
    case 'SIGN_IN_SUCCESS': {
      if (state.status !== 'authenticating') {
        return state;
      }
      return { status: 'authenticated', user: event.user, error: null };
    }
    case 'SIGN_IN_FAILURE': {
      if (state.status !== 'authenticating') {
        return state;
      }
      return { status: 'unauthenticated', user: null, error: event.error };
    }
    case 'REFRESH_START': {
      if (state.status !== 'authenticated') {
        return state;
      }
      return { ...state, status: 'refreshing', error: null };
    }
    case 'REFRESH_SUCCESS': {
      if (state.status !== 'refreshing') {
        return state;
      }
      return { ...state, status: 'authenticated', error: null };
    }
    case 'REFRESH_FAILURE': {
      return { status: 'expired', user: state.user, error: 'session-expired' };
    }
    case 'USER_UPDATED': {
      if (state.status !== 'authenticated' && state.status !== 'refreshing') {
        return state;
      }
      return { ...state, user: event.user };
    }
    case 'SIGNED_OUT': {
      return { status: 'unauthenticated', user: null, error: null };
    }
    default: {
      return state;
    }
  }
}

import { useReducer, useCallback } from 'react';
import {
  sendCiaChat,
} from '@/services/api/pilot';
import {
  classifyCiaError,
  toCiaMessage,
  type CiaGate,
  type CiaMessageVM,
} from '@/services/adapters/cia';

export type CiaChatState = {
  messages: CiaMessageVM[];
  sessionId: string | null;
  sending: boolean;
  gate: CiaGate;
  gateMessage: string | null;
};

type Action =
  | { type: 'ADD_USER_MESSAGE'; message: CiaMessageVM }
  | { type: 'ADD_PENDING_CIA'; message: CiaMessageVM }
  | { type: 'REPLACE_PENDING_CIA'; pendingId: string; message: CiaMessageVM; sessionId: string }
  | { type: 'MARK_FAILED'; userId: string; gate: CiaGate; gateMessage: string }
  | { type: 'CLEAR_FAILED'; userId: string }
  | { type: 'RESET_GATE' };

const initialState: CiaChatState = {
  messages: [],
  sessionId: null,
  sending: false,
  gate: null,
  gateMessage: null,
};

function reducer(state: CiaChatState, action: Action): CiaChatState {
  switch (action.type) {
    case 'ADD_USER_MESSAGE': {
      return {
        ...state,
        gate: null,
        gateMessage: null,
        sending: true,
        messages: [...state.messages, action.message],
      };
    }
    case 'ADD_PENDING_CIA': {
      return {
        ...state,
        messages: [...state.messages, action.message],
      };
    }
    case 'REPLACE_PENDING_CIA': {
      const filtered = state.messages.filter(m => m.id !== action.pendingId);
      return {
        ...state,
        sending: false,
        sessionId: action.sessionId,
        gate: null,
        gateMessage: null,
        messages: [...filtered, action.message],
      };
    }
    case 'MARK_FAILED': {
      const updated = state.messages.map(m =>
        m.id === action.userId ? { ...m, failed: true } : m
      ).filter(m => !m.pending);
      
      return {
        ...state,
        sending: false,
        messages: updated,
        gate: action.gate,
        gateMessage: action.gateMessage,
      };
    }
    case 'CLEAR_FAILED': {
      return {
        ...state,
        gate: null,
        gateMessage: null,
        messages: state.messages.map(m =>
          m.id === action.userId ? { ...m, failed: false } : m
        ),
      };
    }
    case 'RESET_GATE': {
      return {
        ...state,
        gate: null,
        gateMessage: null,
      };
    }
    default:
      return state;
  }
}

export function useCiaChat(opts?: {
  isOnboarding?: boolean;
  starter?: CiaMessageVM;
}): CiaChatState & {
  send(text: string): Promise<void>;
  retry(messageId: string): Promise<void>;
} {
  const starter = opts?.starter;
  const isOnboarding = opts?.isOnboarding;
  const [state, dispatch] = useReducer(reducer, starter, (s) =>
    s ? { ...initialState, messages: [s] } : initialState
  );
  const sessionId = state.sessionId;

  const performSend = useCallback(async (text: string, userId: string) => {
    const pendingId = `cia-pending-${Date.now()}`;

    dispatch({
      type: 'ADD_PENDING_CIA',
      message: {
        id: pendingId,
        role: 'cia',
        text: '',
        pending: true,
      },
    });

    try {
      const response = await sendCiaChat({
        message: text,
        sessionId: sessionId ?? undefined,
        isOnboarding,
      });

      const ciaMessage = toCiaMessage(response);

      dispatch({
        type: 'REPLACE_PENDING_CIA',
        pendingId,
        message: ciaMessage,
        sessionId: response.sessionId,
      });
    } catch (err) {
      const { gate, userMessage } = classifyCiaError(err);
      dispatch({
        type: 'MARK_FAILED',
        userId,
        gate,
        gateMessage: userMessage,
      });
    }
  }, [isOnboarding, sessionId]);

  const send = useCallback(async (text: string) => {
    const userId = `user-${Date.now()}`;
    dispatch({
      type: 'ADD_USER_MESSAGE',
      message: {
        id: userId,
        role: 'user',
        text,
      },
    });

    await performSend(text, userId);
  }, [performSend]);

  const retry = useCallback(async (messageId: string) => {
    const message = state.messages.find(m => m.id === messageId);
    if (!message) {
      return;
    }

    dispatch({ type: 'CLEAR_FAILED', userId: messageId });
    await performSend(message.text, messageId);
  }, [performSend, state.messages]);

  return {
    messages: state.messages,
    sessionId: state.sessionId,
    sending: state.sending,
    gate: state.gate,
    gateMessage: state.gateMessage,
    send,
    retry,
  };
}

  prefs: PrivacyPrefsVM | null;
  status: 'pending' | 'error' | 'success';
  update: (patch: Partial<PrivacyPrefsVM>) => void;
  updating: boolean;
} {
  const queryClient = useQueryClient();

  const query = useQuery<PreferencesResponseDto, ApiError>({
    queryKey: queryKeys.preferences(),
    queryFn: getPreferences,
  });

  const prefs = useMemo(() => {
    if (!query.data) {
      return null;
    }
    return query.data.preferences.privacy;
  }, [query.data]);

  const mutation = useMutation<
    PrivacyPatchResponseDto,
    ApiError,
    Partial<PrivacyPreferencesDto>,
    { previousData?: PreferencesResponseDto }
  >({
    mutationFn: (patch: Partial<PrivacyPreferencesDto>) => updatePrivacy(patch),
    onMutate: async (patch: Partial<PrivacyPreferencesDto>) => {
      await queryClient.cancelQueries({ queryKey: queryKeys.preferences() });

      const previousData = queryClient.getQueryData<PreferencesResponseDto>(queryKeys.preferences());

      if (previousData) {
        const optimisticData: PreferencesResponseDto = {
          ...previousData,
          preferences: {
            ...previousData.preferences,
            privacy: {
              ...previousData.preferences.privacy,
              ...patch,
            },
          },
        };
        queryClient.setQueryData<PreferencesResponseDto>(queryKeys.preferences(), optimisticData);
      }

      return { previousData };
    },
    onError: (_error, _patch, context) => {
      if (context?.previousData) {
        queryClient.setQueryData<PreferencesResponseDto>(queryKeys.preferences(), context.previousData);
      }
    },
    onSettled: () => {
      queryClient.invalidateQueries({ queryKey: queryKeys.preferences() });
    },
  });

  return {
    prefs,
    status: query.status,
    update: (patch: Partial<PrivacyPrefsVM>) => mutation.mutate(patch),
    updating: mutation.isPending,
  };
}

## Screen contracts

**Sign-in**: form calls `useSession().signIn(email,password)`; `authenticating` → disabled button + inline progress; failure → inline `ErrorState`-style message (server message passthrough, e.g. "Invalid email or password"); success → router.replace `/(auth)/onboarding` when `useOnboardingStatus` says `!isComplete`, else `/(tabs)/today`. When arriving with `status === 'expired'` show the canon notice: **"Your session ended. Sign in to continue."** Remove any "pilot mode" bypass buttons — sign-in is real now (if a demo entry existed, it dies here; document removal in a code comment).

**Session gating** (`app/index.tsx` + tabs layout): `booting` → minimal splash (SkeletonState hero); `authenticated|refreshing` → tabs; `unauthenticated|expired` → sign-in. Expired must not dead-end inside tabs: tab layout redirects when status becomes expired.

**Cia screen**: `useCiaChat({ starter })` — keep existing starter message copy (as a `CiaMessageVM`). Composer: send → optimistic user bubble + pending Cia bubble ("Cia is thinking…" subtle); response replaces pending; failure marks the user message failed with a tap-to-retry affordance. Gates render as a distinct card ABOVE the composer per `gate`: `feature_disabled`/`plan_upgrade_required` → `LockedFeatureState` (title "Cia coaching is locked", detail = gateMessage, no fake unlock action — CTA disabled with label "Available with a plan" since no purchase flow exists this batch); `credits_exhausted` → honest info card (not locked-styled); transient → retry affordance on the failed bubble. NEVER a fabricated Cia reply. Purple accents allowed here (Cia surface).

**Cia onboarding screen**: keep the quick/deep choice; on selection start `useCiaChat({ isOnboarding: true })` conversation inline (same optimistic pattern); a "Continue to Today" `BtnPrimary` appears once at least one Cia reply arrived OR the user skips (`BtnGhost` "Skip for now"); both route `/(tabs)/today`. Show `useOnboardingStatus` step summary honestly if available (steps done/pending); no fake progress.

**Data-controls screen**:
1. Privacy section: toggles for `shareProgressWithCoach`, `showInLeaderboards`, `allowAnonymousDataForResearch` via `usePrivacyPrefs` (optimistic, rollback on error). Visibility rows for `profileVisibility`/`healthProfileVisibility` can render read-only value chips this batch.
2. Export section: `useDataExport` — run → summary list (per-table exported/total, amber "truncated" chip when `truncated`), honest empty tables shown as 0/0; result kept in memory only (no file write this batch) with a note "Full JSON export available on request".
3. Delete section (destructive, two-step): step 1 `BtnDanger`-style "Delete my account" → expands confirm card; step 2 requires typing the EXACT phrase `DELETE MY ACCOUNT` in a `GlassPillInput`-style field + password field (always shown; helper text "Leave empty if you signed up with Google/Apple") + final confirm button disabled until phrase matches exactly. Copy MUST state: "This is immediate and permanent. There is no undo." On success: `useSession().signOut()`-equivalent local cleanup + route to sign-in.
4. Gaps + gated: "Consent history" row → honest `HonestNullState`-style inline note "Not available yet — coming when the backend ships it" (no fake log). `GATED_FEATURES` list rendered with lock glyphs + honest reasons.
5. Session section: "Clear local session" (existing behavior) → `signOut()`.

**Me screen**: keep hub layout; stats row switches to `useGamification` StatsVM (level/xp/streak real numbers); sign-out button calls `useSession().signOut()`; nav buttons to Life Areas / Fitness / Data Controls unchanged.
