/**
 * Shared motion constants (BIOS-004 ADR-6: "one source of truth prevents
 * drift across GLM-authored components").
 *
 * Framework-agnostic ON PURPOSE — this file exports only plain numbers,
 * tuples, and objects. It does NOT import 'react-native' or
 * 'react-native-reanimated'.
 *
 * Verified live 2026-07-09: `import { Easing } from 'react-native-reanimated'`
 * throws during Vitest collection under this project's test config
 * (vitest.config.ts -> environment: 'node'):
 *   "Cannot find module '.../react-native-worklets/lib/module/initializers/
 *    initializers' imported from .../react-native-worklets/lib/module/index.js"
 * reanimated's entrypoint reaches into the native worklets runtime, which
 * does not exist under plain Node. src/**\/*.test.ts (vitest.config.ts
 * `include`) collects ANY test that transitively imports this file, so a
 * top-level reanimated import here would break every future test that
 * touches motion.ts. Component files that actually animate (running under
 * Metro/the RN runtime, never under Vitest) build the real Reanimated easing
 * curve at the call site:
 *
 *   import { Easing } from 'react-native-reanimated';
 *   import { MotionEasing } from '@/constants/motion';
 *   const curve = Easing.bezier(...MotionEasing.physical);
 *
 * Reduced motion:
 * Gate every animated component on the existing EXPORTED hook —
 * `useReducedMotion` from '@/components/balencia/auth-inputs' (verified
 * live: exported at that path). NOTE: '@/components/balencia/states.tsx'
 * has its own private, non-exported duplicate of the same hook —
 * pre-existing drift, not introduced or fixed by this file; import the
 * auth-inputs one.
 * When `useReducedMotion()` is true: skip straight to the FINAL static state
 * — no sweep, draw, stagger, spring, or breathe — but text value updates and
 * haptics still fire (canon: "text updates and haptics still occur").
 * Breathing collapses to a static glow (no pulse). Animate `transform` and
 * `opacity` ONLY, never color/blur/shadow (ADR-6, binding).
 */

  sweepShort: 250,
